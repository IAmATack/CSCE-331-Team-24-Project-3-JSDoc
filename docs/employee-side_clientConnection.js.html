/**
 * Simple client-side connection helper to call the server endpoints.
 * Exports a ClientConnection class and a default instance.
 * Methods: getMenu(), getCashiers(), createOrder(order)
 */

export class ClientConnection {
  constructor(baseUrl) {
    // Always use Render server by default
    const RENDER_URL = 'https://team-24-project-3-1.onrender.com';
    
    let defaultBase;
    
    // Priority 1: Explicit baseUrl parameter
    if (baseUrl) {
      defaultBase = baseUrl;
    }
    // Priority 2: localStorage override (for testing - set localStorage.setItem('API_BASE_URL', 'https://...'))
    else if (typeof window !== 'undefined' && window.localStorage) {
      const storedBase = window.localStorage.getItem('API_BASE_URL');
      if (storedBase) {
        defaultBase = storedBase;
      }
    }
    
    // Priority 3: Environment variable (set at build time in Create React App)
    if (!defaultBase && typeof process !== 'undefined' && process.env && process.env.REACT_APP_API_BASE) {
      defaultBase = process.env.REACT_APP_API_BASE;
    }
    
    // Priority 4: Default to Render URL
    if (!defaultBase) {
      defaultBase = RENDER_URL;
    }
    
    this.base = (baseUrl || defaultBase).replace(/\/+$|\s+/g, ''); // trim trailing slashes/spaces
    // Initialize employee ID; in Node, default to 1
    if (typeof window !== 'undefined' && window.localStorage) {
      this.id = window.localStorage.getItem('employeeID') || 1;
    } else {
      this.id = 1;
    }
    // Cookie storage for Node.js environment (browsers handle cookies automatically)
    this.cookies = new Map();
    this.isNode = typeof window === 'undefined' && typeof process !== 'undefined';
    // Session token for API authentication when cookies don't work
    // Try to load from localStorage first (persists across page reloads)
    if (typeof window !== 'undefined' && window.localStorage) {
      this.sessionToken = window.localStorage.getItem('sessionToken') || null;
      if (this.sessionToken) {
        console.log('🔑 Loaded session token from localStorage');
      }
    } else {
      this.sessionToken = null;
    }
    
    // Log the detected base URL for debugging
    console.log('🔧 ClientConnection initialized:');
    console.log('   Base URL:', this.base);
    if (typeof window !== 'undefined' && window.localStorage && window.localStorage.getItem('API_BASE_URL')) {
      console.log('   ⚠️ Using localStorage override:', window.localStorage.getItem('API_BASE_URL'));
    }
  }

  async request(path, { method = 'GET', headers = {}, body, timeoutMs = 30000 } = {}) {
    // universal fetch wrapper
    const fullPath = path.startsWith('http://') || path.startsWith('https://')
      ? path
      : `${this.base}${path.startsWith('/') ? '' : '/'}${path}`;

    console.log(`🌐 Making ${method} request to: ${fullPath}`);
    if (body) {
      console.log(`📦 Request body:`, typeof body === 'string' ? body : JSON.stringify(body));
    }

    const controller = typeof AbortController !== 'undefined' ? new AbortController() : null;
    
    // Build headers with cookies for Node.js environment
    const requestHeaders = { 'Accept': 'application/json', ...headers };
    
    // Use session token if available (works in both Node.js and browser environments)
    // This is a fallback when cookies don't work or for cross-origin scenarios
    if (this.sessionToken) {
      requestHeaders['Authorization'] = `Bearer ${this.sessionToken}`;
      console.log(`🔑 Using session token for authentication`);
    }
    
    // In Node.js, manually handle cookies since fetch doesn't persist them automatically
    if (this.isNode && this.cookies.size > 0) {
      const cookieString = Array.from(this.cookies.entries())
        .map(([name, value]) => `${name}=${value}`)
        .join('; ');
      requestHeaders['Cookie'] = cookieString;
      console.log(`🍪 Sending cookies: ${cookieString.substring(0, 50)}...`);
    }
    
    const opts = {
      method,
      headers: requestHeaders,
      credentials: 'include', // Include cookies for session management (browser)
      signal: controller ? controller.signal : undefined
    };

    if (body !== undefined) {
      opts.body = typeof body === 'string' ? body : JSON.stringify(body);
      opts.headers['Content-Type'] = opts.headers['Content-Type'] || 'application/json';
    }

    let timer;
    // timer controller to abort fetch after timeoutMs
    try {
      if (controller) timer = setTimeout(() => controller.abort(), timeoutMs);
      const res = await fetch(fullPath, opts);
      if (timer) clearTimeout(timer);

      // Extract and store cookies from response (Node.js only)
      if (this.isNode && res.headers) {
        let setCookieHeader = null;
        let cookiesArray = [];
        
        // Method 1: Try getSetCookie() - Node.js 18+ native fetch
        if (typeof res.headers.getSetCookie === 'function') {
          try {
            cookiesArray = res.headers.getSetCookie();
            if (cookiesArray && cookiesArray.length > 0) {
              console.log('🍪 Extracted', cookiesArray.length, 'cookie(s) via getSetCookie()');
            }
          } catch (e) {
            console.log('⚠️ getSetCookie() failed:', e.message);
          }
        }
        
        // Method 2: Try accessing raw headers (undici format)
        if (cookiesArray.length === 0 && res.headers.raw) {
          try {
            const rawSetCookie = res.headers.raw['set-cookie'];
            if (rawSetCookie) {
              cookiesArray = Array.isArray(rawSetCookie) ? rawSetCookie : [rawSetCookie];
              console.log('🍪 Extracted', cookiesArray.length, 'cookie(s) via raw headers');
            }
          } catch (e) {
            console.log('⚠️ raw headers access failed:', e.message);
          }
        }
        
        // Method 3: Try get() method (may return comma-separated string)
        if (cookiesArray.length === 0 && typeof res.headers.get === 'function') {
          try {
            setCookieHeader = res.headers.get('set-cookie');
            if (setCookieHeader) {
              // Split by comma, but be careful - cookies can have commas in values
              // Simple approach: split and parse each
              cookiesArray = [setCookieHeader];
              console.log('🍪 Extracted cookie via get()');
            }
          } catch (e) {
            console.log('⚠️ get() method failed:', e.message);
          }
        }
        
        // Parse and store cookies
        if (cookiesArray.length > 0) {
          for (const cookie of cookiesArray) {
            if (typeof cookie === 'string') {
              // Parse cookie: name=value; path=/; httpOnly; ...
              // Handle both full cookie string and just name=value
              const parts = cookie.split(';');
              const nameValue = parts[0].trim();
              const match = nameValue.match(/^([^=]+)=(.+)$/);
              if (match) {
                const [, name, value] = match;
                this.cookies.set(name.trim(), value.trim());
                console.log(`🍪 Stored cookie: ${name.trim()}=${value.trim().substring(0, 30)}...`);
              }
            }
          }
        } else if (this.isNode) {
          console.log('⚠️ No Set-Cookie header found. Available headers:');
          // Debug: log all header names
          if (res.headers && typeof res.headers.forEach === 'function') {
            const headerNames = [];
            res.headers.forEach((value, key) => {
              headerNames.push(key);
            });
            console.log('   Headers:', headerNames.join(', '));
          } else if (res.headers && typeof res.headers.keys === 'function') {
            const headerNames = Array.from(res.headers.keys());
            console.log('   Headers:', headerNames.join(', '));
          }
        }
      }

      const contentType = res.headers.get && res.headers.get('content-type');
      const text = await res.text();

      let data;
      if (contentType && contentType.includes('application/json')) {
        try { data = JSON.parse(text); } catch (e) { data = text; }
      } else {
        data = text;
      }

      if (!res.ok) {
        const err = new Error(`Request failed ${res.status} ${res.statusText}`);
        err.status = res.status;
        err.body = data;
        throw err;
      }

      return data;
    } catch (err) {
      if (err.name === 'AbortError') {
        const timeoutErr = new Error(`Request timed out after ${timeoutMs}ms to ${fullPath}`);
        timeoutErr.name = 'TimeoutError';
        throw timeoutErr;
      }
      
      // Handle network errors (CORS, connection refused, etc.)
      if (err instanceof TypeError && err.message.includes('fetch')) {
        const networkErr = new Error(
          `Network error: Failed to connect to ${fullPath}. ` +
          `This could be due to CORS, server being down, or network issues. ` +
          `Original error: ${err.message}`
        );
        networkErr.name = 'NetworkError';
        networkErr.originalError = err;
        networkErr.url = fullPath;
        throw networkErr;
      }
      
      // Re-throw other errors as-is
      throw err;
    }
  }

   * Create a new order

   * @param {boolean} [wantsReceiptEmail=false] - Whether the customer wants to receive receipt via email
   * @returns {Promise<Object>} Order creation result with orderID

  async createOrder(order, email = null, wantsReceiptEmail = false) {
    try {
      console.log('🔄 API Call: createOrder()');
      // Accepts order object as described by server logic
      // Include email and wantsReceiptEmail in the request body
      const requestBody = {
        ...order,
        ...(email && { email }),
        // Preserve flag from order if present; otherwise use param
        wantsReceiptEmail:

            ? Boolean(order.wantsReceiptEmail)
            : Boolean(wantsReceiptEmail)
      };
      const data = await this.request('/createOrder', { method: 'POST', body: requestBody });
      // server returns { success: true, orderID, order }
      console.log('✅ API Success: createOrder()');
      return data;

      console.error('❌ API Failed: createOrder()', err.message || err);

  /**
   * Authenticate with Google OAuth
   * @param {string} accessToken - Google OAuth access token
   * @returns {Promise<Object>} Login result with employee info if successful
   */
  async googleAuth(accessToken) {
    try {
      console.log('🔄 API Call: googleAuth()');
      console.log('📍 Backend URL:', this.base);
      console.log('📍 Frontend origin:', typeof window !== 'undefined' ? window.location.origin : 'N/A');
      console.log('📍 Full URL:', `${this.base}/googleAuth`);
      
      const data = await this.request('/googleAuth', {
        method: 'POST',
        body: { token: accessToken }
      });
      if (data.success && data.sessionToken) {
        this.sessionToken = data.sessionToken;
        if (typeof window !== 'undefined' && window.localStorage) {
          window.localStorage.setItem('sessionToken', data.sessionToken);
          console.log('🔑 Session token stored in localStorage for future requests');
        } else {
          console.log('🔑 Session token stored for future requests');
        }
      }
      console.log('✅ API Success: googleAuth()');
      return data;
    } catch (err) {
      console.error('❌ API Failed: googleAuth()', err);
      console.error('📍 Attempted URL:', `${this.base}/googleAuth`);
      console.error('📍 Frontend origin:', typeof window !== 'undefined' ? window.location.origin : 'N/A');
      console.error('📍 Error type:', err.name);
      console.error('📍 Error message:', err.message);
      console.error('📍 Full error:', err);
      
      if (err.name === 'NetworkError' || (err.message && err.message.includes('fetch'))) {
        const frontendOrigin = typeof window !== 'undefined' ? window.location.origin : 'N/A';
        throw new Error(
          `Cannot connect to backend server at ${this.base}. ` +
          `Frontend origin: ${frontendOrigin}. ` +
          `Make sure the backend is running and CORS is configured to allow ${frontendOrigin}.`
        );
      }
      throw err;
    }
  }
