/**
 * Middleware to check if user is authenticated (logged in)
 * Supports both cookie-based sessions and token-based authentication via Authorization header
 * @param {Request} req - Express request object
 * @param {Response} res - Express response object
 * @param {Function} next - Express next middleware function
 */
export async function requireAuth(req, res, next) {
  /**
   * Require an authenticated session or token.
   * Attaches user context to the request when available.
   * @param {Request} req
   * @param {Response} res
   * @param {Function} next
   */
  const requestPath = `${req.method} ${req.path}`;
  
  // If session exists but employee is missing, try to reload from store
  if (req.session && !req.session.employee && req.sessionStore && typeof req.sessionStore.get === 'function') {
    req.sessionStore.get(req.sessionID, (err, session) => {
      if (!err && session && session.employee) {
        req.session.employee = session.employee;
        // Only log session reloads as they're infrequent and indicate a potential issue
        console.log(`✅ [AUTH] Session reloaded for ${requestPath} - User: ${session.employee.username || session.employee.email}`);
      }
      continueAuth();
    });
    return;
  }
  
  function continueAuth() {
    // Check if session exists (cookie-based auth)
    if (req.session && req.session.employee) {
      if (!req.session.employee.isActive) {
        console.error(`❌ [AUTH] Inactive account attempted ${requestPath} - User: ${req.session.employee.username || req.session.employee.email}`);
        return res.status(403).json({ 
          error: 'Your account is inactive. Please contact your manager.',
          errorCode: ErrorCodes.AUTH_ACCOUNT_INACTIVE_COOKIE,
          errorCodeName: 'AUTH_ACCOUNT_INACTIVE_COOKIE',
          authenticated: false 
        });
      }
      // Removed verbose success logging - only log failures
      return next();
    }

    // Check for token-based auth via Authorization header
    const authHeader = req.headers.authorization;
    if (authHeader && authHeader.startsWith('Bearer ')) {
      const sessionToken = authHeader.substring(7);
      // Removed verbose attempt logging - only log failures
      
      try {
        const sessionStore = req.sessionStore;
        if (sessionStore && typeof sessionStore.get === 'function') {
          sessionStore.get(sessionToken, (err, session) => {
            if (err || !session || !session.employee) {
              console.error(`❌ [AUTH] Token auth failed for ${requestPath} - ${err ? 'Error: ' + err.message : 'Session not found or no employee data'}`);
              const errorCode = err ? ErrorCodes.AUTH_SESSION_EXPIRED : ErrorCodes.AUTH_TOKEN_INVALID;
              return res.status(401).json({ 
                error: 'Authentication required. Please log in.',
                errorCode,
                errorCodeName: getErrorCodeName(errorCode),
                authenticated: false 
              });
            }
            
            if (!session.employee.isActive) {
              console.error(`❌ [AUTH] Inactive account attempted ${requestPath} - User: ${session.employee.username || session.employee.email}`);
              return res.status(403).json({ 
                error: 'Your account is inactive. Please contact your manager.',
                errorCode: ErrorCodes.AUTH_ACCOUNT_INACTIVE_TOKEN,
                errorCodeName: 'AUTH_ACCOUNT_INACTIVE_TOKEN',
                authenticated: false 
              });
            }
            
            req.session = session;
            req.sessionID = sessionToken;
            // Removed verbose success logging - only log failures
            return next();
          });
          return;
        }
      } catch (error) {
        console.error(`❌ [AUTH] Token auth error for ${requestPath}:`, error.message);
        // Fall through to 401
      }
    } else {
      console.error(`❌ [AUTH] No authentication found for ${requestPath}`);
      console.error(`   Session exists: ${!!req.session}`);
      console.error(`   Session has employee: ${!!(req.session && req.session.employee)}`);
      console.error(`   Authorization header: ${authHeader ? 'Present' : 'Missing'}`);
      console.error(`   Cookies: ${req.headers.cookie || 'None'}`);
    }

    // No valid authentication found
    return res.status(401).json({ 
      error: 'Authentication required. Please log in.',
      errorCode: ErrorCodes.AUTH_NO_SESSION,
      errorCodeName: 'AUTH_NO_SESSION',
      authenticated: false 
    });
  }
  
  continueAuth();
}

/**
 * Require manager role for protected endpoints.
 * @param {Request} req
 * @param {Response} res
 * @param {Function} next
 */
export function requireManager(req, res, next) {
  /**
   * Ensure the caller has manager privileges.
   * @param {Request} req
   * @param {Response} res
   * @param {Function} next
   */
  if (!req.session || !req.session.employee) {
    return res.status(401).json({ 
      error: 'Authentication required',
      errorCode: ErrorCodes.AUTH_REQUIRED,
      errorCodeName: 'AUTH_REQUIRED',
      authenticated: false 
    });
  }

  if (req.session.employee.role !== 'manager') {
    console.log(`❌ Authorization failed: ${req.session.employee.username || req.session.employee.email} attempted manager-only operation`);
    return res.status(403).json({ 
      error: 'Manager access required. This operation is restricted to managers only.',
      errorCode: ErrorCodes.AUTH_NOT_MANAGER,
      errorCodeName: 'AUTH_NOT_MANAGER',
      authenticated: true,
      authorized: false
    });
  }

  // Removed verbose success logging - only log failures
  next();
}
