import { Link } from 'react-router-dom';
import React from 'react';
import "./KitchenView.css";
import Dropdown from "../components/Dropdown/Dropdown.js";
import DropdownItem from "../components/Dropdown/DropdownItem.js";
import client from './clientConnection.js';

/**
 * KitchenView component: shows current incomplete orders for kitchen staff.
 *
 * - Polls the backend every 30 seconds for undelivered orders and merges
 *   any new incomplete orders into the displayed list.
 * - Allows marking orders as completed which updates the backend and UI.
 *
 * @returns {JSX.Element}
 */
function KitchenView() {

    /** Array of incomplete orders to display; each item is an order object from the API */
    const [orderList, setOrderList] = React.useState([]);

    /**
     * Poll for new incomplete orders on mount and every 30 seconds thereafter.
     * @returns {void}
     */
    React.useEffect(() => {
        refreshOrders();

        const interval = setInterval(refreshOrders, 30000);
        return () => clearInterval(interval);
    }, []);

    /**
     * Fetch latest incomplete orders and merge them into the `orderList` state.
     * Makes multiple backend calls: one to obtain undelivered order ids, then
     * individual calls to retrieve full order details for each id.
     * @async
     * @returns {Promise<void>}
     */
    const refreshOrders = async () => {
        try {
            // Step 1 — Get ALL orders (like before)
            const allOrders = await client.getReport(24);
            console.log(allOrders);

            // Filter only the undelivered ones
            const undeliveredIds = allOrders.data
                .filter(order => order.delivered === false)
                .map(order => order.id);
            console.log(undeliveredIds);

            if (!undeliveredIds.length) return;

            // Step 2 — For each ID, fetch the detailed incomplete order
            const orderPromises = undeliveredIds.map(id =>
                client.getIncompleteOrderById(Number(id))
            );

            const orderResults = await Promise.all(orderPromises);

            // Extract the actual order objects from API responses
            // Server returns { success: true, order: [row] }
            const newOrders = orderResults
                .map(res => Array.isArray(res?.order) ? res.order[0] : (res?.order || res?.data || null))
                .filter(order => order && order.order_id);

            // Step 3 — Merge while avoiding duplicates
            setOrderList(prev => {
                const existingIds = new Set(prev.map(o => o.order_id));

                const onlyNew = newOrders.filter(
                    order => !existingIds.has(order.order_id)
                );

                return [...prev, ...onlyNew];
            });

        } catch (error) {
            console.error("Failed to fetch incomplete orders:", error);
        }
    };

    /**
     * Mark the order as completed (delivered) in the backend and remove it from UI.
     * @param {number} orderID
     * @async
     * @returns {Promise<void>}
     */
    const markOrderAsCompleted = async (orderID) => {
        try {
            await client.genericTableSet({
                table: 'orders',
                column: 'delivered',
                value: true,
                id: orderID
            });
        } catch (error) {
            console.error("Failed to mark order as completed:", error);
            alert("Failed to mark order as completed: " + error.message);
        }

        // Remove the completed order from UI
        setOrderList(prevItems => prevItems.filter(item => item.order_id !== orderID));
    };

    /** Render the kitchen orders UI */
    return (
  <div>
    <h2 id="kitchen-view-heading">Kitchen View</h2>

    {/* Live region for new orders or updates */}
    <div
      aria-live="polite"
      aria-atomic="true"
      className="ordersContainer"
      role="region"
      aria-labelledby="kitchen-view-heading"
    >
      {orderList.map((order) => (
        <div
          key={order.order_id}
          className="orderCard"
          role="group"
          aria-labelledby={`order-${order.order_id}-heading`}
        >
          <h3 id={`order-${order.order_id}-heading`}>
            Order #{order.order_id} {order.order_name}
          </h3>

          {/* Time elapsed with aria-label */}
          <p aria-label={`Time elapsed for order ${order.order_id}`}>
            {order.timeelapsed.days ? `${order.timeelapsed.days}d ` : ""}
            {order.timeelapsed.hours ? `${order.timeelapsed.hours}h ` : ""}
            {order.timeelapsed.minutes ? `${order.timeelapsed.minutes}m ` : ""}
            {order.timeelapsed.seconds ? `${order.timeelapsed.seconds}s` : ""}
          </p>

          <div
            role="list"
            aria-label={`Items for order ${order.order_id}`}
          >
            {order.line_items.map((item) => (
              <div key={item.id} className="item" role="listitem">
                <span>{`${item.line_item_quantity}x ${item.line_item_name}`}</span>

                {item.meal_items.length > 0 && (
                  <div
                    role="list"
                    aria-label={`Sub-items for ${item.line_item_name}`}
                  >
                    {item.meal_items.map((subItem) => (
                      <div key={subItem.id} className="subItem" role="listitem">
                        {subItem.meal_item_name}
                      </div>
                    ))}
                  </div>
                )}
              </div>
            ))}
          </div>

          <button
            className="completeButton"
            onClick={() => markOrderAsCompleted(order.order_id)}
            aria-label={`Mark order ${order.order_id} as completed`}
          >
            Complete
          </button>
        </div>
      ))}
    </div>
  </div>
);

} export default KitchenView;
