import { Link } from 'react-router-dom';
import React from 'react';
import "./Menu.css";
import Dropdown from "../components/Dropdown/Dropdown.js";
import DropdownItem from "../components/Dropdown/DropdownItem.js";
import client from './clientConnection.js';

/**
 * Menu component: provides UI and logic to manage menu items and meal options.
 *
 * - Loads menu and inventory data via `client`.
 * - Supports viewing/updating meal options and menu items, creating new menu items,
 *   and composing menu ingredients from inventory entries.
 *
 * @returns {JSX.Element}
 */
function Menu() {
    /** Meal option field labels used in dropdowns */
    const mealOptionFields = ["Name", "Price"];

    /**
     * Meal option names (filtered by mealtype === 'M')
     * @type {Array<string>}
     */
    const [mealOptions, setMealOptions] = React.useState([]);

    /**
     * Selected meal option details shown in the UI
     * @type {{id:number,itemname:string,price:number}}
     */
    const [selectedMealOption, updateSelectedMealOption] = React.useState ({
        id: -1,
        itemname: "Select a Meal Option",
        price: 0.0,
    });

    /** Which field on the selected meal option is being edited */
    const [selectedMealOptionField, updateSelectedMealOptionField] = React.useState("Field to Update");

    /** Temporary new value for the selected meal option field */
    const [newMealOptionValue, setNewMealOptionValue] = React.useState("");

    /** Menu item field labels used in dropdowns */
    const menuItemFields = ["Name", "Price", "Active?"];

    /**
     * Menu item names (excluding meal options)
     * @type {Array<string>}
     */
    const [menuItems, setMenuItems] = React.useState([]);

    /**
     * Selected menu item details shown in the UI
     * @type {{id:number,itemname:string,price:number,isactive:boolean}}
     */
    const [selectedMenuItem, updateSelectedMenuItem] = React.useState ({
        id: -1,
        itemname: "Select a Menu Item",
        price: 0.0,
        isactive: false
    });

    /** Which field on the selected menu item is being edited */
    const [selectedMenuItemField, updateSelectedMenuItemField] = React.useState("Field to Update");

    /** Temporary new value for the selected menu item field */
    const [newMenuItemValue, setNewMenuItemValue] = React.useState("");

    /**
     * Select a meal option by name. Triggers the effect that loads full details.
     * @param {string} name
     * @returns {void}
     */
    const onMealOptionSelect = (name) => {
        updateSelectedMealOption({
            itemname: name,
        })
    }

    /** Select which field of the meal option will be edited */
    const onMealOptionFieldSelect = (name) => {
        updateSelectedMealOptionField(name);
    }

    /** Load meal option names (mealtype === 'M') on mount */
    React.useEffect(() => {
        const fetchItems = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "menu",
                    columns: ["itemname", "mealtype"],
                });
                const meals = data.filter(item => item.mealtype === "M");
                const names = meals.map(item => item.itemname);
                setMealOptions(names);
            } catch (error) {
                console.error("An error occurred:", error.message);
                setMealOptions(["Unable to load items"]);
            }
        };

        fetchItems();
    }, []);

    /**
     * When `selectedMealOption.itemname` changes, fetch full details for that meal option
     * @returns {void}
     */
    React.useEffect(() => {
        if (!selectedMealOption.itemname || selectedMealOption.itemname === "Select a Meal Option") return; // do nothing if no selection

        const fetchItemDetails = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "menu",
                    columns: ["id", "itemname", "price"],
                });

                // Filter the result for the selected Item
                const itm = data.find(e => e.itemname === selectedMealOption.itemname);

                if (itm) {
                    updateSelectedMealOption({
                        id: itm.id ?? -1,
                        itemname: itm.itemname ?? "",
                        price: itm.price ?? 0.0,
                    });
                }
            } catch (error) {
                console.error("Failed to fetch item details:", error.message);
            }
        };

        fetchItemDetails();
    }, [selectedMealOption.itemname]);

    /**
     * Validate and submit an update for the selected meal option
     * @async
     * @returns {Promise<void>}
     */
    const onUpdateMealOptionClick = async () => {
        if (!selectedMealOption.itemname || !selectedMealOptionField || newMealOptionValue === "") {
            alert("Please select an item, a field, and enter a new value");
            return;
        }

        try {
            // Map field names to database column names
            const columnMap = {
                "Name": "itemname",
                "Price": "price",
            };

            const column = columnMap[selectedMealOptionField];

            if (!column) throw new Error("Invalid field selected");
            
            // Explicit type conversion if necessary
            let valueToSet;
            if (column === "price") {
                valueToSet = Number(newMealOptionValue);
            } else {
                valueToSet = newMealOptionValue;
            }

            await client.genericTableSet({
                table: "menu",
                column: column,
                value: valueToSet,
                id: selectedMealOption.id 
            });

            alert("Item updated successfully!");

            const updatedData = await client.genericTableGet({
                table: "menu",
                columns: ["id", "itemname", "price"],
                id: selectedMealOption.id
            });

            if (updatedData && updatedData.length > 0) {
                const itm = updatedData[0];
                updateSelectedMealOption({
                    id: itm.id ?? -1,
                    itemname: itm.itemname ?? "",
                    price: itm.price ?? 0.0,
                });
            }
        } catch (error) {
            console.error("Failed to update item:", error);
            alert("Failed to update item: " + error.message);
        }
    }

    /**
     * Select a menu item by name; triggers the effect that loads full details.
     * @param {string} name
     * @returns {void}
     */
    const onMenuItemSelect = (name) => {
        updateSelectedMenuItem({
            itemname: name,
            isactive: false
        })
    }

    /** Select which field of the menu item will be edited */
    const onMenuItemFieldSelect = (name) => {
        updateSelectedMenuItemField(name);
    }

    /** Load menu item names (non-meal options) on mount */
    React.useEffect(() => {
        const fetchItems = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "menu",
                    columns: ["itemname", "mealtype"],
                });
                const meals = data.filter(item => item.mealtype !== "M");
                console.log(JSON.stringify(meals));
                const names = meals.map(item => item.itemname);
                console.log(JSON.stringify(names));
                setMenuItems(names);
            } catch (error) {
                console.error("An error occurred:", error.message);
                setMenuItems(["Unable to load items"]);
            }
        };

        fetchItems();
    }, []);

    /**
     * When `selectedMenuItem.itemname` changes, fetch full details for that menu item
     * @returns {void}
     */
    React.useEffect(() => {
        if (!selectedMenuItem.itemname || selectedMenuItem.itemname === "Select a Menu Item") return; // do nothing if no selection
        const fetchItemDetails = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "menu",
                    columns: ["id", "itemname", "price", "isactive"],
                });

                // Filter the result for the selected Item
                const itm = data.find(e => e.itemname === selectedMenuItem.itemname);

                if (itm) {
                    updateSelectedMenuItem({
                        id: itm.id ?? -1,
                        itemname: itm.itemname ?? "",
                        price: itm.price ?? 0.0,
                        isactive: itm.isactive ?? false
                    });
                }
            } catch (error) {
                console.error("Failed to fetch item details:", error.message);
            }
        };

        fetchItemDetails();
    }, [selectedMenuItem.itemname]);

    /**
     * Validate and submit an update for the selected menu item
     * @async
     * @returns {Promise<void>}
     */
    const onUpdateMenuItemClick = async () => {
        if (!selectedMenuItem.itemname || !selectedMenuItemField || newMenuItemValue === "") {
            alert("Please select an item, a field, and enter a new value");
            return;
        }

        try {
            // Map field names to database column names
            const columnMap = {
                "Name": "itemname",
                "Price": "price",
                "Active?": "isactive"
            };

            const column = columnMap[selectedMenuItemField];

            if (!column) throw new Error("Invalid field selected");
            
            // Explicit type conversion if necessary
            let valueToSet;
            if (column === "price") {
                valueToSet = Number(newMenuItemValue);
            } else {
                valueToSet = newMenuItemValue;
            }

            await client.genericTableSet({
                table: "menu",
                column: column,
                value: valueToSet,
                id: selectedMenuItem.id 
            });

            alert("Item updated successfully!");

            const updatedData = await client.genericTableGet({
                table: "menu",
                columns: ["id", "itemname", "price", "isactive"],
                id: selectedMenuItem.id
            });

            if (updatedData && updatedData.length > 0) {
                const itm = updatedData[0];
                updateSelectedMenuItem({
                    id: itm.id ?? -1,
                    itemname: itm.itemname ?? "",
                    price: itm.price ?? 0.0,
                    isactive: itm.isactive ?? false
                });
            }
        } catch (error) {
            console.error("Failed to update item:", error);
            alert("Failed to update item: " + error.message);
        }
    }


    /** Menu item type options for the Create form */
    const itemTypes = ["Appetizer", "Drink", "Entree", "Side"];

    /** New menu item form fields */
    const [newItemName, setNewItemName] = React.useState("");
    const [newItemPrice, setNewItemPrice] = React.useState(0);
    const [newItemIsActive, setNewItemIsActive] = React.useState(false);
    const [newItemType, setNewItemType] = React.useState("");

    /** Inventory ingredient list used for selecting ingredients */
    const [ingredients, setIngredients] = React.useState([]);

    /** Ingredients added to the new menu item (array of {name,quantity}) */
    const [addedIngredients, setAddedIngredients] = React.useState([]);

    /** Load inventory item names for ingredient selection on mount */
    React.useEffect(() => {
        const fetchItems = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "inventory",
                    columns: ["itemname"]
                });

                setIngredients(data);
            } catch (error) {
                console.error("An error occurred:", error.message);
                setIngredients(["Unable to load items"]);
            }
        };

        fetchItems();
    }, []);

    /** Add a blank ingredient row to the Create form */
    const addIngredient = () => {
        setAddedIngredients(prev => [
            ...prev,
            { name: "", quantity: 0 }
        ]);
    };

    /** Update an added ingredient's field (name or quantity) */
    const updateIngredient = (index, field, value) => {
        const updated = [...addedIngredients];
        updated[index] = {
            ...updated[index],
            [field]: value
        };
        setAddedIngredients(updated);
    };

    /**
     * Save a new menu item and its ingredients to the database
     * @async
     * @returns {Promise<void>}
     */
    const onSaveNewItemClick = async () => {
        console.log("Hi");
        if (!newItemName || newItemPrice === 0 || !newItemIsActive || !newItemType) return;
        console.log("1");
        for (let i of addedIngredients) {
            if (!i.name || i.quantity === 0) {
                console.log(i.name);
                console.log(i.quantity);
                console.log(i);
                return;
            }
        }
        console.log("2");

        const typeMap = {
            "Appetizer": 'A',
            "Drink": 'D',
            "Entree": 'E',
            "Side": 'S'
        };
        const typeToSet = typeMap[newItemType];

        try {
            const result = await client.genericTableCreate({
                table: "menu",
                values: {
                    itemname: newItemName,
                    price: Number(newItemPrice),
                    mealtype: typeToSet,
                    isactive: Boolean(newItemIsActive)
                }
            });

            for (let i of addedIngredients) {
                const data = await client.genericTableGet({
                    table: "inventory",
                    columns: ["id", "itemname"],
                });
                const itemEntry = data.find(e => e.itemname === i.name);

                await client.genericTableCreate({
                    table: "menuingredients",
                    values: {
                        menu_id: result.id,
                        inventory_id: itemEntry.id,
                        quantity: Number(i.quantity)
                    }
                });
            }

            alert("Item added successfully!");
        } catch (error) {
            console.error("Failed to add item:", error);
            alert("Failed to add item: " + error.message);
        }

    }

 return (
  <div>
    {/* Navigation */}
    <nav
      style={{ padding: '10px', backgroundColor: '#f0f0f0' }}
      role="navigation"
      aria-label="Manager navigation"
    >
      <Link to="/manager" style={{ margin: '10px' }}>Manager Home</Link>
      <Link to="/employees" style={{ margin: '10px' }}>Employees</Link>
      <Link to="/inventory" style={{ margin: '10px' }}>Inventory</Link>
      <Link to="/menu" style={{ margin: '10px' }}>Menu</Link>
      <Link to="/reports" style={{ margin: '10px' }}>Reports</Link>
    </nav>

    <div className="grid-container">
      {/* Meal Options & Menu Items */}
      <div className="row" role="region" aria-labelledby="meal-menu-heading">
        <h2 id="meal-menu-heading" className="sr-only">Meal Options and Menu Items Management</h2>
        <div className="nested-grid-container">

          {/* Meal Options */}
          <div className="column" role="region" aria-labelledby="meal-options-heading">
            <h2 id="meal-options-heading">Meal Options</h2>

            <Dropdown buttonText={selectedMealOption.itemname} aria-label="Select Meal Option">
              {mealOptions.map((itm, index) => (
                <DropdownItem key={index} onClick={() => onMealOptionSelect(itm)}>
                  {itm}
                </DropdownItem>
              ))}
            </Dropdown>

            <table aria-label="Selected Meal Option details">
              <tbody>
                <tr>
                  <td>Name:</td>
                  <td>{selectedMealOption.itemname}</td>
                </tr>
                <tr>
                  <td>Price:</td>
                  <td>{selectedMealOption.price}</td>
                </tr>
              </tbody>
            </table>

            <Dropdown buttonText={selectedMealOptionField} aria-label="Select field to update for Meal Option">
              {mealOptionFields.map((field) => (
                <DropdownItem key={field} onClick={() => onMealOptionFieldSelect(field)}>
                  {field}
                </DropdownItem>
              ))}
            </Dropdown>
            <label>
              New Value:
              <input
                aria-label="New value for selected Meal Option field"
                value={newMealOptionValue}
                onChange={(e) => setNewMealOptionValue(e.target.value)}
              />
            </label>
            <button className="defaultButton" onClick={onUpdateMealOptionClick}>
              Save
            </button>
          </div>

          {/* Menu Items */}
          <div className="column" role="region" aria-labelledby="menu-items-heading">
            <h2 id="menu-items-heading">Menu Items</h2>

            <Dropdown buttonText={selectedMenuItem.itemname} aria-label="Select Menu Item">
              {menuItems.map((itm, index) => (
                <DropdownItem key={index} onClick={() => onMenuItemSelect(itm)}>
                  {itm}
                </DropdownItem>
              ))}
            </Dropdown>

            <table aria-label="Selected Menu Item details">
              <tbody>
                <tr>
                  <td>Name:</td>
                  <td>{selectedMenuItem.itemname}</td>
                </tr>
                <tr>
                  <td>Price:</td>
                  <td>{selectedMenuItem.price}</td>
                </tr>
                <tr>
                  <td>Active?</td>
                  <td>{selectedMenuItem.isactive.toString() || ""}</td>
                </tr>
              </tbody>
            </table>

            <Dropdown buttonText={selectedMenuItemField} aria-label="Select field to update for Menu Item">
              {menuItemFields.map((field) => (
                <DropdownItem key={field} onClick={() => onMenuItemFieldSelect(field)}>
                  {field}
                </DropdownItem>
              ))}
            </Dropdown>
            <label>
              New Value:
              <input
                aria-label="New value for selected Menu Item field"
                value={newMenuItemValue}
                onChange={(e) => setNewMenuItemValue(e.target.value)}
              />
            </label>
            <button className="defaultButton" onClick={onUpdateMenuItemClick}>
              Save
            </button>
          </div>

        </div>
      </div>

      {/* Create New Menu Item */}
      <div className="row" role="region" aria-labelledby="create-menu-item-heading">
        <h2 id="create-menu-item-heading">Create a New Menu Item</h2>

        <div className="addItemRow">
          <label>
            Name:
            <input
              aria-label="New menu item name"
              value={newItemName}
              onChange={(e) => setNewItemName(e.target.value)}
            />
          </label>

          <label>
            Price:
            <input
              aria-label="New menu item price"
              type="number"
              value={newItemPrice}
              onChange={(e) => setNewItemPrice(Number(e.target.value))}
              min="0"
              max="100"
              step=".25"
            />
          </label>

          <Dropdown buttonText={newItemIsActive || "Active?"} aria-label="Set active status for new menu item">
            {["true", "false"].map((val) => (
              <DropdownItem key={val} onClick={() => setNewItemIsActive(val)}>
                {val}
              </DropdownItem>
            ))}
          </Dropdown>

          <Dropdown buttonText={newItemType || "Select Type"} aria-label="Select type for new menu item">
            {itemTypes.map((type) => (
              <DropdownItem key={type} onClick={() => setNewItemType(type)}>
                {type}
              </DropdownItem>
            ))}
          </Dropdown>
        </div>

        <button className="defaultButton" onClick={addIngredient} aria-label="Add a new ingredient">
          Add Ingredient
        </button>

        {addedIngredients.map((ingredient, index) => (
          <div className="addItemRow" key={index} role="group" aria-labelledby={`ingredient-${index}-heading`}>
            <h4 id={`ingredient-${index}-heading`}>Ingredient #{index + 1}</h4>

            <Dropdown
              buttonText={ingredient.name || "Select Ingredient"}
              aria-label={`Select name for ingredient #${index + 1}`}
            >
              {ingredients.map((itm, idx) => (
                <DropdownItem key={idx} onClick={() => updateIngredient(index, "name", itm.itemname)}>
                  {itm.itemname}
                </DropdownItem>
              ))}
            </Dropdown>

            <label>
              Quantity:
              <input
                type="number"
                aria-label={`Quantity for ingredient #${index + 1}`}
                value={ingredient.quantity}
                onChange={(e) => updateIngredient(index, "quantity", e.target.value)}
                min="0"
                max="1000"
                step="1"
              />
            </label>
          </div>
        ))}

        <button
          className="defaultButton"
          style={{ marginTop: "20px" }}
          onClick={onSaveNewItemClick}
          aria-label="Save the new menu item"
        >
          Save New Menu Item
        </button>
      </div>
    </div>
  </div>
);

}

export default Menu;
