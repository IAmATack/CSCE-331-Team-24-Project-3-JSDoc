  /**
   * Validate employee credentials and create a session.
   */
  app.post('/validateEmployeeLogin', async (req, res) => {
    try {
      const { username, password, role } = req.body;
      
      if (!username) {
        console.error('Validation failed: Username is required');
        return res.status(400).json({ error: 'Username is required' });
      }
      if (!password) {
        console.error('Validation failed: Password is required');
        return res.status(400).json({ error: 'Password is required' });
      }
      if (role !== 0 && role !== 1) {
        console.error('Validation failed: Role must be 0 or 1');
        return res.status(400).json({ error: 'Role must be 0 (cashier) or 1 (manager)' });
      }

      const result = await logic.validateEmployeeLogin(username, password, role);
      
      // If login successful, create session
      if (result.success && result.employee) {
        // IMPORTANT: Set employee data BEFORE any session operations
        // This ensures the data is part of the session object
        req.session.employee = {
          id: result.employee.id,
          name: result.employee.name,
          username: result.employee.username,
          role: result.employee.role,
          isActive: result.employee.isActive,
        };
        
        // Force the session to be saved by marking it as modified
        // Some session stores require explicit modification tracking
        req.session.touch();
        
        // Save the session explicitly and wait for it before sending response
        // CRITICAL: Use sessionStore.set directly instead of req.session.save()
        // This ensures we're using the PostgreSQL store directly
        const sessionDataToSave = {
          cookie: req.session.cookie,
          employee: req.session.employee
        };
        
        await new Promise((resolve, reject) => {
          // Use a callback to ensure save completes
          // Also set a timeout to catch if save never completes
          const saveTimeout = setTimeout(() => {
            console.error('❌ Session save timed out after 5 seconds!');
            reject(new Error('Session save timeout'));
          }, 5000);
          
          // Try direct sessionStore.set first
          if (req.sessionStore && typeof req.sessionStore.set === 'function') {
            req.sessionStore.set(req.sessionID, sessionDataToSave, (err) => {
              clearTimeout(saveTimeout);
              if (err) {
                console.error('❌ Direct sessionStore.set() FAILED:', err);
                // Fall back to req.session.save()
                req.session.save((saveErr) => {
                  if (saveErr) {
                    console.error('❌ req.session.save() also FAILED:', saveErr);
                    reject(saveErr);
                    return;
                  }
                  resolve();
                });
                return;
              }
              
              resolve();
            });
          } else {
            // Fall back to req.session.save()
            req.session.save((err) => {
              clearTimeout(saveTimeout);
              if (err) {
                console.error('❌ Error saving session:', err);
                reject(err);
                return;
              }
              resolve();
            });
          }
        });
        
        // Return session token for API clients (Node.js scripts, etc.)
        // This allows authentication via Authorization header when cookies don't work
        result.sessionToken = req.sessionID;
      }
      
      // CRITICAL: Ensure cookie is set by explicitly touching the session
      // This forces express-session to set the cookie in the response
      if (result.success && req.sessionID) {
        // Force session to be marked as modified so cookie is set
        req.session.touch();
      }
      
      res.json(result);
    } 
    catch (e) { 
      console.error('API Error [POST /validateEmployeeLogin]:', e.message || e);
      res.status(500).json({ error: e.message || 'An unknown error occurred' }); 
    }
  });

  /**
   * Authenticate via Google OAuth token and create a session.

  app.post('/googleAuth', async (req, res) => {
    try {
      const { token } = req.body;
      
      if (!token) {
        console.error('Validation failed: Google token is required');
        return res.status(400).json({ error: 'Google token is required' });
      }

      // Verify Google token
      const googleUser = await verifyGoogleToken(token);

      // Find employee in database by email
      const employee = await findEmployeeByEmail(googleUser.email, logic);
      
      if (!employee) {
        console.error('Employee not found for email:', googleUser.email);
        return res.status(404).json({ 
          success: false,
          error: 'No employee account found with this email. Please contact your manager.' 
        });
      }

      if (!employee.isActive) {
        console.error('Employee account is inactive:', googleUser.email);
        return res.status(403).json({ 
          success: false,
          error: 'Your account is inactive. Please contact your manager.' 
        });
      }

      // Create session
      req.session.employee = {
        id: employee.id,
        name: employee.name,
        username: employee.username,
        email: employee.email,
        role: employee.role,
        isActive: employee.isActive,
        googleId: googleUser.googleId,
      };

      // Touch the session to mark it as modified (required for some session stores)
      req.session.touch();
      
      // Save the session explicitly and wait for it before sending response
      await new Promise((resolve, reject) => {
        req.session.save((err) => {
          if (err) {
            console.error('❌ Error saving Google auth session:', err);
            reject(err);
          } else {
            resolve();
          }
        });
      });

      res.json({
        success: true,
        message: 'Login successful',
        employee: {
          id: employee.id,
          name: employee.name,
          username: employee.username,
          email: employee.email,
          role: employee.role,
          isActive: employee.isActive,
        },
        sessionToken: req.sessionID // Also return session token for API clients
      });
    } 
    catch (e) { 
      console.error('API Error [POST /googleAuth]:', e.message || e);
      res.status(500).json({ 
        success: false,
        error: e.message || 'An unknown error occurred' 
      }); 
    }
  });

  /**
   * Generic table create: insert a row with type validation (requires auth).
   */
  app.post('/genericTableCreate', requireAuth, async (req, res) => {
    try {
      const { table, values } = req.body;
      
      if (!table) {
        const errorCode = ErrorCodes.VAL_MISSING_TABLE;
        const error = '[VALIDATION] Table parameter is required';
        console.error(`❌ Validation failed: ${error}`);
        return res.status(400).json({ 
          success: false,
          error,
          errorCode,
          errorCodeName: getErrorCodeName(errorCode),
          errorType: 'VALIDATION',
          details: 'The "table" field is required in the request body'
        });
      }
      if (!values || typeof values !== 'object' || Array.isArray(values)) {
        const errorCode = ErrorCodes.VAL_INVALID_VALUES;
        const error = '[VALIDATION] Values parameter is required and must be an object';
        console.error(`❌ Validation failed: ${error}`);
        return res.status(400).json({ 
          success: false,
          error,
          errorCode,
          errorCodeName: getErrorCodeName(errorCode),
          errorType: 'VALIDATION',
          details: `The "values" field must be an object with column-value pairs (got: ${typeof values}${Array.isArray(values) ? ' (array)' : ''})`
        });
      }

      const data = await logic.genericTableCreate(table, values);
      res.json({ success: true, data });
    } 
    catch (e) { 
      // Extract error code and type from error object
      const errorCode = e.errorCode || 'UNKNOWN_ERROR';
      const errorType = e.errorType || (e.message && e.message.startsWith('[VALIDATION]') ? 'VALIDATION' : 
                                        e.message && e.message.startsWith('[DATABASE]') ? 'DATABASE' : 'UNKNOWN');
      
      // Determine status code based on error type
      let statusCode = 500;
      if (errorType === 'VALIDATION' || e.message?.startsWith('[VALIDATION]')) {
        statusCode = 400;
      } else if (e.message && (e.message.includes('does not exist') || e.message.includes('not found'))) {
        statusCode = 404;
      } else if (errorType === 'DATABASE' || e.message?.startsWith('[DATABASE]')) {
        // Database errors - check for specific constraint violations
        if (errorCode === ErrorCodes.DB_UNIQUE_VIOLATION || e.message?.includes('Duplicate entry') || e.message?.includes('already exists')) {
          statusCode = 409; // Conflict
        } else if (errorCode === ErrorCodes.DB_NOT_NULL_VIOLATION || e.message?.includes('NOT NULL') || e.message?.includes('required')) {
          statusCode = 400; // Bad Request
        } else {
          statusCode = 500; // Internal Server Error
        }
      }
      
      // Log error
      console.error(`API Error [POST /genericTableCreate]:`, e.message || 'An unknown error occurred');
      if (e.dbCode) {
        console.error(`   PostgreSQL Code: ${e.dbCode} | Constraint: ${e.dbConstraint || 'N/A'} | Column: ${e.dbColumn || 'N/A'}`);
      }
      
      // Build response object with all error information
      const errorResponse = { 
        success: false,
        error: e.message || 'An unknown error occurred',
        errorCode,
        errorCodeName: getErrorCodeName(errorCode),
        errorType,
        details: e.message || 'No additional details available'
      };
      
      // Include database-specific error details if available
      if (e.dbCode) {
        errorResponse.dbCode = e.dbCode;
      }
      if (e.dbDetail) {
        errorResponse.dbDetail = e.dbDetail;
      }
      if (e.dbHint) {
        errorResponse.dbHint = e.dbHint;
      }
      if (e.dbConstraint) {
        errorResponse.dbConstraint = e.dbConstraint;
      }
      if (e.dbColumn) {
        errorResponse.dbColumn = e.dbColumn;
      }
      
      res.status(statusCode).json(errorResponse); 
    }
  });
