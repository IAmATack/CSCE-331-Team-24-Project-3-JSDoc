import { Link } from 'react-router-dom';
import React from 'react';
import "./Reports.css";
import "react-datepicker/dist/react-datepicker.css";
import Dropdown from "../components/Dropdown/Dropdown.js";
import DropdownItem from "../components/Dropdown/DropdownItem.js";
import DatePicker from "react-datepicker";
import client from './clientConnection.js';

/**
 * Reports component: UI for selecting and generating pre-defined reports.
 *
 * - Loads the list of available reports from the backend via `client.getReportList()`.
 * - For date-range reports, exposes `fromDate` and `toDate` pickers.
 * - Calls `client.getReport(id, params?)` to fetch report data and renders it in a table.
 *
 * @returns {JSX.Element}
 */
function Reports() {
    /** List of available reports (loaded on mount) @type {Array<Object>} */
    const [reportList, setReportList] = React.useState([]);

    /**
     * Currently selected report
     * @type {{name:string,description:string,id:number}}
     */
    const [selectedReport, setSelectedReport] = React.useState({
        name: "",
        description: "",
        id: 0
    })

    /** From-date for date-range reports */
    const [fromDate, setFromDate] = React.useState(new Date());

    /** To-date for date-range reports */
    const [toDate, setToDate] = React.useState(new Date());

    /** Whether date pickers should be shown for the selected report */
    const [isDateVisible, setIsDateVisible] = React.useState(false);

    /** Rows returned by the last generated report */
    const [reportTableData, setReportTableData] = React.useState([]);

    /**
     * Handle user selecting a report from the dropdown.
     * Shows date pickers only for reports that support date ranges (IDs 30 and 31).
     * @param {{id:number,name:string,description:string}} report
     * @returns {void}
     */
    const handleReportSelect = (report) => {
        setSelectedReport(report);
        (report.id === 30 || report.id === 31) ? setIsDateVisible(true) : setIsDateVisible(false);
    }

    /**
     * Load available reports on component mount.
     * @returns {void}
     */
    React.useEffect(() => {
        const fetchReportList = async () => {
            try {
                const data = await client.getReportList();
                console.log(data)
                const reports = data.map(item => ({
                        id: item.id,
                        name: item.name,
                        description: item.description,
                        active: item.active
                    })).filter(report => report.active === true);

                console.log(reports[0]);
                console.log(reports[0].name);
                setReportList(reports);
            } catch (error) {
                console.error("An error occurred:", error.message);
                setReportList(["Unable to load reports"]);
            }
        };

        fetchReportList();
    }, []);

    /**
     * Generate the selected report. For date-range reports (IDs 30,31) the
     * `fromDate` and `toDate` are validated and passed to the backend.
     * @async
     * @returns {Promise<void>}
     */
    const generateReport = async () => {
        if (selectedReport.id === 0) {
            alert("Please select a report.");
            return;
        } if ((selectedReport.id === 30 || selectedReport.id === 31) && (fromDate > toDate)) {
            alert("Please ensure the From Date is before the To Date.");
            return;
        }
        
        let data;
        if(selectedReport.id === 30 || selectedReport.id === 31) {
            try {
                console.log(fromDate.toJSON());
                data = await client.getReport(selectedReport.id, {
                    from: fromDate.toJSON(), 
                    to: toDate.toJSON()
            });
            } catch (error) {
                    console.error("An error occurred:", error.message);
                    alert(["Unable to generate report"]);
                    return;
            }
        } else {
            try {
                data = await client.getReport(selectedReport.id);
            } catch (error) {
                    console.error("An error occurred:", error.message);
                    alert(["Unable to generate report"]);
                    return;
            } 
        }

        console.log(JSON.stringify(data));
        setReportTableData(data.data);
    }

    return (
    <div>
        <nav
        style={{ padding: '10px', backgroundColor: '#f0f0f0' }}
        role="navigation"
        aria-label="Main navigation"
        >
        <Link to="/manager" style={{ margin: '10px' }}>Manager Home</Link>
        <Link to="/employees" style={{ margin: '10px' }}>Employees</Link>
        <Link to="/inventory" style={{ margin: '10px' }}>Inventory</Link>
        <Link to="/menu" style={{ margin: '10px' }}>Menu</Link>
        <Link to="/reports" style={{ margin: '10px' }}>Reports</Link>
        </nav>
        <div role="region" aria-labelledby="report-controls-heading">
        <h2 id="report-controls-heading" className="sr-only">Report Controls</h2>
        <div className="controls">

        <Dropdown buttonText={selectedReport.name || "Select a Report"} aria-label="Select report">
            {reportList.map((data) => (
            <DropdownItem
                key={data.id}
                onClick={() => handleReportSelect(data)}
                aria-label={`Select report ${data.name}`}
            >
                {data.name}
            </DropdownItem>
            ))}
        </Dropdown>

        {isDateVisible && (
            <label>
            From Date: 
            <DatePicker
                selected={fromDate}
                onChange={(date) => setFromDate(date)}
                aria-label="Select start date"
            />
            </label>
        )}

        {isDateVisible && (
            <label>
            To Date: 
            <DatePicker
                selected={toDate}
                onChange={(date) => setToDate(date)}
                aria-label="Select end date"
            />
            </label>
        )}

        <button
            className="defaultButton"
            onClick={generateReport}
            aria-label="Generate report"
        >
            Generate Report
        </button>
        </div>

        {selectedReport.description && (
        <div className="reportDescription" role="region" aria-label="Report description">
            {selectedReport.description}
        </div>
        )}

        <div className="reportTableContainer" role="region" aria-label="Report table">
        <div className="reportTableScroll">
            {reportTableData.length > 0 && (
            <table className="reportTable">
                <caption className="sr-only">
                {selectedReport.name ? `${selectedReport.name} Report` : 'Report Table'}
                </caption>
                <thead>
                <tr>
                    {Object.keys(reportTableData[0]).map((key) => (
                    <th key={key} scope="col">{key}</th>
                    ))}
                </tr>
                </thead>
                <tbody>
                {reportTableData.map((row, rowIndex) => (
                    <tr key={rowIndex}>
                    {Object.keys(row).map((colKey) => (
                        <td key={colKey}>
                        {typeof row[colKey] === "boolean" ? row[colKey].toString() : row[colKey]}
                        </td>
                    ))}
                    </tr>
                ))}
                </tbody>
            </table>
            )}
        </div>
        </div>
        </div>
    </div>
    );
}

export default Reports;
