  /**
 * Formats a date value into `"YYYY-MM-DD HH:mm:SS"` for report generation.
 *
 * Accepts:
 *  - JavaScript Date objects
 *  - ISO date strings (e.g., "2023-05-10T14:30:00Z")
 *  - "YYYY-MM-DD HH:mm:SS" strings (returned unchanged)
 *  - "YYYY-MM-DD" strings (assumes start of day)
 *
 * @param {Date|string|null} date - The date to format.
 * @returns {string|null} The formatted date string or null if input is falsy.
 */
  formatDate(date) {
      if (!date) return null;
      if (date instanceof Date) {
        return date.toISOString().slice(0, 19).replace('T', ' ');
      }
      if (typeof date === 'string') {
        if (date.includes('T') || date.includes('Z')) {
          const parsedDate = new Date(date);
          if (!isNaN(parsedDate.getTime())) {
            return parsedDate.toISOString().slice(0, 19).replace('T', ' ');
          }
        }
        if (date.match(/\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}/)) {
          return date;
        }
        if (date.match(/^\d{4}-\d{2}-\d{2}$/)) {
          return date + ' 00:00:00';
        }
      }
      return date;
    };

  /**
 * @deprecated This function uses legacy plaintext password validation.
 *            Use genericTableGet() instead.
 * 
 * Validates a manager's login credentials by comparing the provided 
 * password with the stored password in the database.
 *
 * @param {string} username - The manager's username used in the DB query.
 * @param {string} password - The plaintext password to validate.
 * @returns {boolean} True if the username exists and the password matches; otherwise false.
 */
  async validateLoginManagers({username, password}) {
    console.log("Validating login for " + username);
    const result = await pool.query("SELECT _password FROM managers WHERE username=$1;",
      [username]
    );

    if(result.rows.length === 0){
      console.log("Username not found");
      return false;
    }

    const databasePassword = result.rows[0]._password;
    return databasePassword===password;
  }
  
  /**
 * Retrieves schema information (table name, column name, and data type)
 * for all tables in the public schema.
 *
 * @returns {Promise<Array<{table_name: string, column_name: string, data_type: string}>>}
 *   An array of objects describing each column in the database.
 */
  /**
   * Retrieve schema (public tables) with column names and data types.
   * @returns {Promise<Array>} Rows with table_name, column_name, data_type.
   */
  async getTableSchema() {
    console.log("Fetching: Database schema");
    
    const result = await pool.query(`
      SELECT 
        table_name,
        column_name,
        data_type
      FROM 
        information_schema.columns
      WHERE 
        table_schema = 'public'
      ORDER BY 
        table_name, ordinal_position;
    `);
    
    console.log("Retrieved: Database schema");
    return result.rows;
  }

  // Report 8: Inventory Usage By Month
  async inventoryUsageByMonth() {
    console.log("Fetching: Inventory usage by month");
    const result = await pool.query(`
      SELECT 
        TO_CHAR(DATE_TRUNC('month', o.timeOfOrder), 'YYYY-MM') AS month,
        i.itemName,
        ROUND(SUM(mi.quantity * ing.quantity), 2) AS totalUsed
      FROM orders o
      JOIN lineItems li ON o.id = li.orderID
      JOIN mealItems mi ON li.itemID = mi.lineItemID
      JOIN menuIngredients ing ON mi.menuID = ing.menu_id
      JOIN inventory i ON ing.inventory_id = i.id
      GROUP BY month, i.itemName
      ORDER BY month, totalUsed DESC;
    `);
    console.log("Retrieved: Inventory usage by month");
    return result.rows;
  }

  // Report 27: Master Orders (Not On List)
  // Function to return all orders connected with the mealItems and lineItems tables
  async getOrdersMaster() {
    console.log("Fetching: Master list of orders");

    const result = await pool.query(`
      WITH meal_agg AS (
      SELECT
          mi.lineItemID,
          JSON_AGG(
              JSON_BUILD_OBJECT(
                  'meal_item_id', mi.id,
                  'meal_item_name', m2.itemName,
                  'meal_item_quantity', mi.quantity,
                  'meal_item_price', m2.price,
                  'meal_item_total', mi.quantity * m2.price
              )
          ) AS meal_items
      FROM mealItems mi
      LEFT JOIN menu m2 ON m2.id = mi.menuID
      GROUP BY mi.lineItemID
      ),

      line_agg AS (
          SELECT
              li.orderID,
              JSON_AGG(
                  JSON_BUILD_OBJECT(
                      'line_item_id', li.itemID,
                      'line_item_name', m1.itemName,
                      'line_item_quantity', li.quantity,
                      'line_item_price', li.price,
                      'line_item_total', li.quantity * li.price,
                      'meal_items', COALESCE(ma.meal_items, '[]'::json)
                  )
              ) AS line_items
          FROM lineItems li
          LEFT JOIN menu m1 ON m1.id = li.itemType
          LEFT JOIN meal_agg ma ON ma.lineItemID = li.itemID
          GROUP BY li.orderID
      )

      SELECT
          o.id AS order_id,
          o.timeOfOrder,
          o.paymentType,
          o.price AS order_total,
          o.cashierID,
          o.delivered,
          o.name,
          COALESCE(la.line_items, '[]'::json) AS line_items
      FROM orders o
      LEFT JOIN line_agg la ON la.orderID = o.id
      ORDER BY o.id;`
    );

    console.log("Retrieved: Master list of orders");

    return result.rows;
  }

  // Report 30: Product Usage Report
  async generateProductUsageReport(from = null, to = null) {
    /**
     * Compute total quantity used per inventory item for a date range.
     * @param {string|Date|null} from
     * @param {string|Date|null} to
     * @returns {Promise<Array<{itemname:string,totalquantity:number}>>}
     */
    console.log("🔄 Generating Product Usage Report");
    console.log("   Date params:", from, to);

    let fromDateStr = formatDate(from);
    let toDateStr = formatDate(to);
    
    // Default to wide range if no dates
    if (!fromDateStr && !toDateStr) {
      fromDateStr = '1970-01-01 00:00:00';
      toDateStr = '2099-12-31 23:59:59';
    } else if (!fromDateStr) {
      fromDateStr = '1970-01-01 00:00:00';
    } else if (!toDateStr) {
      toDateStr = new Date().toISOString().slice(0, 19).replace('T', ' ');
    }

    console.log("   Formatted dates:", fromDateStr, toDateStr);

    const result = await pool.query(`
      SELECT 
        i.itemname,
        COALESCE(SUM(mi.quantity * mi2.quantity), 0) as totalquantity
      FROM inventory i
      LEFT JOIN menuingredients mi2 ON i.id = mi2.inventory_id
      LEFT JOIN mealitems mi ON mi2.menu_id = mi.menuid
      LEFT JOIN lineitems li ON mi.lineitemid = li.itemid
      LEFT JOIN orders o ON li.orderid = o.id
      WHERE ($1::timestamp IS NULL OR o.timeoforder >= $1::timestamp)
        AND ($2::timestamp IS NULL OR o.timeoforder <= $2::timestamp)
      GROUP BY i.id, i.itemname
      HAVING COALESCE(SUM(mi.quantity * mi2.quantity), 0) > 0
      ORDER BY totalquantity DESC, i.itemname;
    `, [fromDateStr, toDateStr]);

    console.log("✅ Product Usage Report generated");
    return result.rows;
  }

  async getIncompleteOrderById(orderId) {
    /**
     * Retrieve an order by id with line and meal item aggregation.
     * Includes elapsed time and delivered flag.
     * @param {number} orderId
     * @returns {Promise<Array<Object>>} Rows for the requested order.
     */
    console.log("Fetching: Incomplete orders");
    const result = await pool.query(`
      WITH meal_agg AS (
      SELECT
          mi.lineItemID,
          JSON_AGG(
              JSON_BUILD_OBJECT(
                  'meal_item_id', mi.id,
                  'meal_item_name', m2.itemName,
                  'meal_item_quantity', mi.quantity,
                  'meal_item_price', m2.price,
                  'meal_item_total', mi.quantity * m2.price
              )
          ) AS meal_items
      FROM mealItems mi
      LEFT JOIN menu m2 ON m2.id = mi.menuID
      GROUP BY mi.lineItemID
      ),

      line_agg AS (
          SELECT
              li.orderID,
              JSON_AGG(
                  JSON_BUILD_OBJECT(
                      'line_item_id', li.itemID,
                      'line_item_name', m1.itemName,
                      'line_item_quantity', li.quantity,
                      'line_item_price', li.price,
                      'line_item_total', li.quantity * li.price,
                      'meal_items', COALESCE(ma.meal_items, '[]'::json)
                  )
              ) AS line_items
          FROM lineItems li
          LEFT JOIN menu m1 ON m1.id = li.itemType
          LEFT JOIN meal_agg ma ON ma.lineItemID = li.itemID
          GROUP BY li.orderID
      )

      SELECT
          o.id AS order_id,
          o.timeOfOrder,
          o.paymentType,
          o.price AS order_total,
          o.cashierID,
          o.delivered,
          o.name,
          DATE_TRUNC('second', NOW() - o.timeOfOrder) AS timeElapsed,
          COALESCE(la.line_items, '[]'::json) AS line_items
      FROM orders o
      LEFT JOIN line_agg la ON la.orderID = o.id
      WHERE id=$1
      ORDER BY timeElapsed DESC;`, [orderId]);
    console.log("Retrieved: Incomplete orders");
    return result.rows;

  }
