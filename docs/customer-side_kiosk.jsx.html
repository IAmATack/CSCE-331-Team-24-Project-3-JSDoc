const MEAL_TYPES = {
  ALC_ENTREE: "ALC_ENTREE",
  ALC_SIDE: "ALC_SIDE",
  BOWL: "BOWL",
  PLATE: "PLATE",
  BIGGER_PLATE: "BIGGER_PLATE",
  UNKNOWN: "UNKNOWN",
};

const MEAL_RULES = {
  [MEAL_TYPES.ALC_ENTREE]: {
    kind: MEAL_TYPES.ALC_ENTREE,
    allowSides: false,
    allowEntrees: true,
    maxSides: 0,
    maxEntrees: 1,
    requireSides: false,
    requireEntrees: true,
    entreeQtyBySize: {
      small: 1,
      medium: 2,
      large: 3,
      default: 1,
    },
  },

  [MEAL_TYPES.ALC_SIDE]: {
    kind: MEAL_TYPES.ALC_SIDE,
    allowSides: true,
    allowEntrees: false,
    maxSides: 1,
    maxEntrees: 0,
    requireSides: true,
    requireEntrees: false,
    sideQtyBySize: {
      medium: 1,
      large: 2,
      default: 1,

  [MEAL_TYPES.BOWL]: {
    kind: MEAL_TYPES.BOWL,
    allowSides: true,
    allowEntrees: true,
    maxSides: 2,
    maxEntrees: 1,
    requireSides: true,

    requireSides: false,
    requireEntrees: false,

function getRulesForSize(sizeItem) {
  if (!sizeItem) return MEAL_RULES[MEAL_TYPES.UNKNOWN];

  const raw = (sizeItem.itemname || "").toLowerCase();

  if (raw.includes("entree") && raw.includes("carte")) {
    return MEAL_RULES[MEAL_TYPES.ALC_ENTREE];
  }

  if (raw.includes("side") && raw.includes("carte")) {
    return MEAL_RULES[MEAL_TYPES.ALC_SIDE];
  }

  if (raw.includes("bowl")) return MEAL_RULES[MEAL_TYPES.BOWL];
  if (raw.includes("plate") && !raw.includes("bigger"))
    return MEAL_RULES[MEAL_TYPES.PLATE];
  if (raw.includes("bigger")) return MEAL_RULES[MEAL_TYPES.BIGGER_PLATE];

  return MEAL_RULES[MEAL_TYPES.UNKNOWN];
}

function getSizeLabelFromName(itemname = "") {
  const raw = itemname.toLowerCase();

  if (raw.startsWith("s_")) return "small";
  if (raw.startsWith("m_")) return "medium";
  if (raw.startsWith("l_")) return "large";

  return "default";
}

function validateMeal(rules, sides, entrees) {
  const type = rules.kind;

  if (type === MEAL_TYPES.ALC_ENTREE) {
    if (entrees.length === 0)
      return { ok: false, message: "Please choose an entrée." };
    if (sides.length > 0)
      return { ok: false, message: "A La Carte Entrées cannot include sides." };
  }

  if (type === MEAL_TYPES.ALC_SIDE) {
    if (sides.length === 0)
      return { ok: false, message: "Please choose a side." };
    if (entrees.length > 0)
      return { ok: false, message: "A La Carte Sides cannot include entrées." };
  }

  if (rules.requireSides && sides.length === 0)
    return { ok: false, message: "Please add at least one side." };

  if (rules.requireEntrees && entrees.length === 0)
    return { ok: false, message: "Please add at least one entrée." };

  if (sides.length > rules.maxSides)
    return {
      ok: false,
      message: `This meal allows up to ${rules.maxSides} side(s).`,
    };

  if (entrees.length > rules.maxEntrees)
    return {
      ok: false,
      message: `This meal allows up to ${rules.maxEntrees} entrée(s).`,
    };

  return { ok: true };
}

function buildMealItems(rules, size, sides, entrees) {
  const type = rules.kind;

  if (type === MEAL_TYPES.ALC_ENTREE) {
    const label = getSizeLabelFromName(size.itemname);
    const qty = rules.entreeQtyBySize?.[label] ?? rules.entreeQtyBySize.default;
    return entrees.map((e) => ({ menuID: e.id, quantity: qty }));
  }

  if (type === MEAL_TYPES.ALC_SIDE) {
    const label = getSizeLabelFromName(size.itemname);
    const qty = rules.sideQtyBySize?.[label] ?? rules.sideQtyBySize.default;
    return sides.map((s) => ({ menuID: s.id, quantity: qty }));
  }

  const sideMealItems =
    sides.length === 1
      ? [{ menuID: sides[0].id, quantity: 1 }]
      : sides.map((s) => ({ menuID: s.id, quantity: 0.5 }));

  const entreeMealItems = entrees.map((e) => ({ menuID: e.id, quantity: 1 }));

  return [...sideMealItems, ...entreeMealItems];
}

/**
 * Formats internal item names into user-friendly display strings for the UI.
 * @param {string} name - Raw database item name.
 * @returns {string} Formatted display name.
 */
function formatSizeDisplayName(name) {
  switch (name) {
    case "Bowl":
      return "Bowl";
    case "Plate":
      return "Plate";
    case "Bigger Plate":
      return "Bigger Plate";
    case "S_entreeCarte":
      return "Small Entrée (A La Carte)";
    case "M_entreeCarte":
      return "Medium Entrée (A La Carte)";
    case "L_entreeCarte":
      return "Large Entrée (A La Carte)";
    case "M_sideCarte":
      return "Medium Side (A La Carte)";
    case "L_sideCarte":
      return "Large Side (A La Carte)";
    default:
      return name;
  }
}

/**
 * Formats appetizer names (replaces underscores with spaces).
 * @param {string} name - Raw item name.
 * @returns {string} Formatted name.

function formatAppDisplayName(name) {
  return name.replace(/_/g, " ");
}

  if (name.startsWith("M_")) return "Medium Soda";
  if (name.startsWith("L_")) return "Large Soda";

}

/**
 * Customer Self-Service Kiosk Component.
 * * Responsibilities:
 * - Provide a touch-friendly interface for customers to build orders.
 * - Manage multi-step navigation flows (Home -> Size -> Sides -> Entrees).
 * - Enforce inventory availability with "Smart Buffer" logic.
 * - Handle customer checkout via Google OAuth or email entry.
 * - Auto-reset session after inactivity.
 * * @component
 */
export default function KioskPOS() {
  const [menu, setMenu] = useState({
    size: [],
    sides: [],
    entrees: [],
    drinks: [],
    appetizers: [],
  });

  const [inventoryMap, setInventoryMap] = useState({});
  const [recipeMap, setRecipeMap] = useState({});

  const [orderItems, setOrderItems] = useState([]);
  const [flow, setFlow] = useState(FLOW.HOME);
  const [step, setStep] = useState(1);

  const [builder, setBuilder] = useState({
    size: null,
    sides: [],
    entrees: [],
  });

  const [selectedApp, setSelectedApp] = useState(null);
  const [selectedDrink, setSelectedDrink] = useState(null);
  const [sessionMessage, setSessionMessage] = useState("");
  const [editingIndex, setEditingIndex] = useState(null);

  /**
   * Effect: Fetches initial menu data from the backend.
   * Categorizes items into sizes, sides, entrees, drinks, and appetizers.
   */
  useEffect(() => {
    async function fetchMenu() {
      try {
        const report = await client.getReport(20);
        const allItems = report.data || report || [];
        const active = allItems.filter((item) => item.isactive);

        setMenu({
          size: active.filter((i) => i.mealtype === "M"),
          sides: active.filter((i) => i.mealtype === "S"),
          entrees: active.filter((i) => i.mealtype === "E"),
          drinks: active.filter((i) => i.mealtype === "D"),
          appetizers: active.filter((i) => i.mealtype === "A"),
        });
      } catch (err) {
        console.error("Error loading menu:", err);
      }
    }

    fetchMenu();
  }, []);

  /**
   * Effect: Polls backend for inventory and recipe data every 5 seconds.
   * Updates local state to enable real-time out-of-stock checking.
   */
  useEffect(() => {
    async function fetchStockData() {
      try {
        // Inventory (Report 23)
        const invReport = await client.getReport(23);
        const inventoryData = invReport.data || [];
        const invMap = {};
        inventoryData.forEach(item => {
            invMap[item.itemname] = Number(item.stock);
        });
        setInventoryMap(invMap);

        // Menu Ingredients (Report 26)
        const recipeReport = await client.getReport(26);
        const recipeData = recipeReport.data || [];
        const rMap = {};
        recipeData.forEach(row => {
            rMap[row.menu_item] = row.ingredients;
        });
        setRecipeMap(rMap);
      } catch (err) {
        console.error("Error updating stock:", err);
      }
    }

    fetchStockData();
    const intervalId = setInterval(fetchStockData, 5000);
    return () => clearInterval(intervalId);
  }, []);

  /**
   * Checks if a specific menu item is out of stock based on current inventory.
   * Applies "Smart Buffer" logic:
   * - Ingredients requiring > 1 unit (bulk) use a strict 1x threshold.
   * - Ingredients requiring <= 1 unit use a 5x safety buffer.
   * * @param {Object} menuItem - The menu item object to check.
   * @returns {boolean} True if the item should be disabled/grayed out.
   */
  const isOutOfStock = (menuItem) => {
    if (!menuItem) return false;

    const ingredients = recipeMap[menuItem.itemname];

    if (!ingredients || ingredients.length === 0) return false;

    for (const ing of ingredients) {
      const currentStock = inventoryMap[ing.ingredient];
      const singleServingReq = Number(ing.quantity);
      const bufferMultiplier = singleServingReq > 1 ? 1 : 5;
      const minimumThreshold = singleServingReq * bufferMultiplier;

      if (currentStock === undefined || currentStock < minimumThreshold) {
        return true;
      }
    }
    return false;
  };

  /**
   * Effect: Monitors user activity.
   * Resets the entire flow and clears order if inactive for INACTIVITY_TIMEOUT_MS.
   */
  useEffect(() => {
    const hasActiveSession =
      orderItems.length > 0 ||
      flow !== FLOW.HOME ||
      builder.size ||
      selectedApp ||
      selectedDrink;

    if (!hasActiveSession) return;

    const timer = setTimeout(() => {
      setOrderItems([]);
      resetFlow();
      setSessionMessage("Session timed out. Starting a new order.");
    }, INACTIVITY_TIMEOUT_MS);

    return () => clearTimeout(timer);
  }, [orderItems, flow, builder, selectedApp, selectedDrink]);

  useEffect(() => {
    if (!sessionMessage) return;

    const timer = setTimeout(() => {
      setSessionMessage("");
    }, 5000);

    return () => clearTimeout(timer);
  }, [sessionMessage]);

  const subtotal = orderItems.reduce(
    (sum, it) => sum + Number(it.price || 0),
    0
  );

  const baseSizes = menu.size.filter((s) => {
    const name = s.itemname.toLowerCase();
    return name === "bowl" || name === "plate" || name === "bigger plate";
  });
  const aLaCarteSizes = menu.size.filter((s) =>
    s.itemname.toLowerCase().includes("carte")
  );

  /**
   * Loads an item from the order list into the appropriate builder flow for editing.
   * Handles both standard meals and A La Carte items correctly.
   * @param {number} idx - Index of the item to edit.
   */
  const editItem = (idx) => {
    const item = orderItems[idx];
    if (!item) return;

    setEditingIndex(idx);

    const alcSizeItem = menu.size.find(
      (x) => x.id === item.menuID && x.itemname.toLowerCase().includes("carte")
    );
    if (alcSizeItem) {
      const rules = getRulesForSize(alcSizeItem);

      const sides = [];
      const entrees = [];

      (item.mealItems || []).forEach((mi) => {
        const menuItem =
          menu.sides.find((s) => s.id === mi.menuID) ||
          menu.entrees.find((e) => e.id === mi.menuID);

        if (!menuItem) return;

        if (rules.kind === MEAL_TYPES.ALC_ENTREE) entrees.push(menuItem);
        if (rules.kind === MEAL_TYPES.ALC_SIDE) sides.push(menuItem);
      });

      setBuilder({
        size: alcSizeItem,
        sides,
        entrees,
      });

      setFlow(FLOW.ALC);
      setStep(1);
      return;
    }

    const sizeItem = menu.size.find(
      (x) => x.id === item.menuID && !x.itemname.toLowerCase().includes("carte") // exclude ALC
    );
    if (sizeItem) {
      const sides = [];
      const entrees = [];

      (item.mealItems || []).forEach((mi) => {
        const menuItem =
          menu.sides.find((s) => s.id === mi.menuID) ||
          menu.entrees.find((e) => e.id === mi.menuID);
        if (!menuItem) return;

        if (menuItem.mealtype === "S") sides.push(menuItem);
        if (menuItem.mealtype === "E") entrees.push(menuItem);
      });

      setBuilder({
        size: sizeItem,
        sides,
        entrees,
      });

      setFlow(FLOW.MEAL);
      setStep(1);
      return;
    }
  };

  /**
   * Initiates the standard meal flow (Bowl/Plate) with a selected size.
   * @param {Object} sizeItem - The container size selected.
   */
  const startMealFlow = (sizeItem) => {
    if (isOutOfStock(sizeItem)) return; 
    setBuilder({
      size: sizeItem,
      sides: [],
      entrees: [],
    });
    setFlow(FLOW.MEAL);
    setStep(1);
  };

  /**
   * Initiates the A La Carte flow.
   * @param {Object} sizeItem - The A La Carte size item (Small Entree, Medium Side, etc.).
   */
  const startALaCarteFlow = (sizeItem) => {
    if (isOutOfStock(sizeItem)) return;
    setBuilder({
      size: sizeItem,
      sides: [],
      entrees: [],
    });
    setFlow(FLOW.ALC);
    setStep(1);
  };

  /**
   * Initiates the Appetizer selection flow.
   * @param {Object} appItem - The selected appetizer.
   */
  const startAppFlow = (appItem) => {
    if (isOutOfStock(appItem)) return;
    setSelectedApp(appItem);
    setFlow(FLOW.APP);
  };

  /**
   * Initiates the Drink selection flow.
   * @param {Object} drinkItem - The selected drink.
   */
  const startDrinkFlow = (drinkItem) => {
    if (isOutOfStock(drinkItem)) return;
    setSelectedDrink(drinkItem);
    setFlow(FLOW.DRINK);
  };

  /**
   * Resets the Kiosk to the Home screen and clears all builder state.
   * Also clears customer checkout inputs.
   */
  const resetFlow = () => {
    setFlow(FLOW.HOME);
    setStep(1);
    setBuilder({ size: null, sides: [], entrees: [] });
    setSelectedApp(null);
    setSelectedDrink(null);
    setEditingIndex(null);
    setShowCheckout(false);
    setCustomerName("");
    setCustomerEmail("");
    setSendReceipt(false);
  };

  /**
   * Toggles a side selection in the meal builder.
   * Enforces maximum side limits defined in MEAL_RULES.
   * @param {Object} item - The side item to toggle.
   */
  const toggleSide = (item) => {
    if (isOutOfStock(item)) return; 
    setBuilder((prev) => {
      const rules = getRulesForSize(prev.size);
      if (!rules.allowSides) return prev;

      const exists = prev.sides.some((s) => s.id === item.id);
      if (exists) {
        return {
          ...prev,
          sides: prev.sides.filter((s) => s.id !== item.id),
        };
      }

      if (prev.sides.length >= rules.maxSides) {
        alert(`You can select up to ${rules.maxSides} sides.`);
        return prev;
      }

      return { ...prev, sides: [...prev.sides, item] };
    });
  };

  /**
   * Toggles an entree selection in the meal builder.
   * Enforces maximum entree limits defined in MEAL_RULES.
   * @param {Object} item - The entree item to toggle.
   */
  const toggleEntree = (item) => {
    if (isOutOfStock(item)) return;
    setBuilder((prev) => {
      const rules = getRulesForSize(prev.size);
      if (!rules.allowEntrees) return prev;

      const exists = prev.entrees.some((e) => e.id === item.id);
      if (exists) {
        return {
          ...prev,
          entrees: prev.entrees.filter((e) => e.id !== item.id),
        };
      }

      if (prev.entrees.length >= rules.maxEntrees) {
        alert(`You can select up to ${rules.maxEntrees} entrées.`);
        return prev;
      }

      return { ...prev, entrees: [...prev.entrees, item] };
    });
  };

  /**
   * Validates the constructed meal and adds it to the cart.
   * Calculates premium upcharges and formats the display name.
   */
  const addBuiltMealToOrder = () => {
    const { size, sides, entrees } = builder;
    if (!size) return;

    const rules = getRulesForSize(size);
    const validation = validateMeal(rules, sides, entrees);
    if (!validation.ok) {
      alert(validation.message);
      return;
    }

    const mealItems = buildMealItems(rules, size, sides, entrees);

    const extraCharge = entrees.reduce(
      (sum, e) => sum + (Number(e.price) > 0 ? Number(e.price) : 0),
      0
    );
    const lineItem = {
      menuID: size.id,
      quantity: 1,
      mealItems,
      itemname: `${formatSizeDisplayName(size.itemname)} – ${[
        ...sides.map((s) => s.itemname),
        ...entrees.map((e) => e.itemname),
      ]
        .filter(Boolean)
        .join(", ")}`,
      price: Number(size.price) + extraCharge,
    };
    if (editingIndex !== null) {
      setOrderItems((prev) =>
        prev.map((it, i) => (i === editingIndex ? lineItem : it))
      );
      setEditingIndex(null);
      resetFlow();
      return;
    }

    setOrderItems((prev) => [...prev, lineItem]);
    resetFlow();
  };
  const addAppToOrder = () => {
    if (!selectedApp) return;
    if (editingIndex !== null) {
      setOrderItems((prev) =>
        prev.map((it, i) =>
          i === editingIndex
            ? {
                menuID: selectedApp.id,
                quantity: 1,
                itemname: formatAppDisplayName(selectedApp.itemname),
                price: selectedApp.price,
              }
            : it
        )
      );
      setEditingIndex(null);
      resetFlow();
      return;
    }
    setOrderItems((prev) => [
      ...prev,
      {
        menuID: selectedApp.id,
        quantity: 1,
        entrees: [],
        itemname: formatAppDisplayName(selectedApp.itemname),
        price: selectedApp.price,
      },
    ]);
    resetFlow();
  };
  const addDrinkToOrder = () => {
    if (!selectedDrink) return;
    if (editingIndex !== null) {
      setOrderItems((prev) =>
        prev.map((it, i) =>
          i === editingIndex
            ? {
                menuID: selectedDrink.id,
                quantity: 1,
                itemname: formatDrinkDisplayName(selectedDrink.itemname),
                price: selectedDrink.price,
              }
            : it
        )
      );
      setEditingIndex(null);
      resetFlow();
      return;
    }
    setOrderItems((prev) => [
      ...prev,
      {
        menuID: selectedDrink.id,
        quantity: 1,
        entrees: [],
        itemname: formatDrinkDisplayName(selectedDrink.itemname),
        price: selectedDrink.price,
      },
    ]);
    resetFlow();
  };
  const removeItem = (idx) => {
    setOrderItems((prev) => prev.filter((_, i) => i !== idx));
  };

  /**
   * Prepares the checkout modal.
   * Validates that the cart is not empty before proceeding.
   */
  const completeOrder = async () => {
    if (orderItems.length === 0) {
      alert("Your order is empty.");
      return;
    }
    // Open checkout modal for name/email / Google auth
    setShowCheckout(true);
  };

  // Checkout modal state
  const [showCheckout, setShowCheckout] = useState(false);
  const [customerName, setCustomerName] = useState("");
  const [customerEmail, setCustomerEmail] = useState("");
  const [checkoutError, setCheckoutError] = useState("");
  const [checkoutLoading, setCheckoutLoading] = useState(false);
  const [sendReceipt, setSendReceipt] = useState(false);
  const [successInfo, setSuccessInfo] = useState(null); // { orderID, points }
  // Clear checkout state when leaving the page/component
  useEffect(() => {
    return () => {
      setShowCheckout(false);
      setCustomerName("");
      setCustomerEmail("");
      setSendReceipt(false);
    };
  }, []);

  // Auto-clear success banner after a short delay
  useEffect(() => {
    if (!successInfo) return;
    const t = setTimeout(() => setSuccessInfo(null), 6000);
    return () => clearTimeout(t);
  }, [successInfo]);

  const googleClientId = process.env.REACT_APP_GOOGLE_CLIENT_ID || "";

  const validateEmail = (email) => {
    if (!email) return false;
    const re = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    return re.test(String(email).toLowerCase());
  };

  /**
   * Validates customer input (Name/Email) and sends the order to the backend.
   * Handles points calculation and receipt requests.
   */
  const submitCheckout = async () => {
    setCheckoutError("");
    if (orderItems.length === 0) return setCheckoutError("Order is empty");
    const hasName = Boolean(customerName && customerName.trim());
    const hasValidEmail = Boolean(
      customerEmail && validateEmail(customerEmail)
    );
    if (sendReceipt && !hasValidEmail) {
      return setCheckoutError("Enter a valid email to receive a receipt");
    }
    if (!hasName && !hasValidEmail) {
      return setCheckoutError("Please sign in or enter your name");
    }
    if (customerEmail && !validateEmail(customerEmail)) {
      return setCheckoutError("Please enter a valid email or leave it blank");
    }

    setCheckoutLoading(true);
    try {
      const payload = {
        email: customerEmail.trim(),
        ...(customerName && { name: customerName.trim() }),
        paymentType: "Card",
        cashierID: null,
        lineItems: orderItems.map((it) => ({
          menuID: it.menuID,
          quantity: it.quantity,
          entree: it.mealItems,
        })),
        wantsReceiptEmail: Boolean(sendReceipt),
      };

      const result = await client.createOrder(payload);
      if (result.success) {
        const pts =
          typeof result.pointsAwarded === "number"
            ? result.pointsAwarded
            : result.order?.pointsAwarded ?? 0;
        const totalPts =
          typeof result.customerPointsTotal === "number"
            ? result.customerPointsTotal
            : result.order?.customerPointsTotal ?? null;
        const oid = result.orderID || result.order?.id || "";
        const emailAttempted =
          typeof result.emailAttempted === "boolean"
            ? result.emailAttempted
            : !!result.order?.emailAttempted;
        const emailPending =
          typeof result.emailPending === "boolean"
            ? result.emailPending
            : !!result.order?.emailPending;
        const emailSent =
          typeof result.emailSent === "boolean"
            ? result.emailSent
            : !!result.order?.emailSent;
        const receiptRequested = Boolean(sendReceipt);
        setSuccessInfo({
          orderID: oid,
          points: pts,
          totalPoints: totalPts,
          emailAttempted,
          emailPending,
          emailSent,
          receiptRequested,
        });
        setOrderItems([]);
        setShowCheckout(false);
        setCustomerName("");
        setCustomerEmail("");
        setSendReceipt(false);
        resetFlow();
      } else {
        setCheckoutError(result.error || result.message || "Order failed");
      }
    } catch (err) {
      setCheckoutError(err.message || "Network error");
    } finally {
      setCheckoutLoading(false);
    }
  };

  /**
   * Callback for successful Google OAuth login.
   * Decodes the JWT token to extract Name and Email.
   * @param {Object} credentialResponse - Response from Google Login API.
   */
  const handleGoogleSuccess = (credentialResponse) => {
    try {
      const token = credentialResponse?.credential;
      if (!token) return setCheckoutError("Google sign-in failed");
      // decode JWT payload to extract email and name
      const base64Url = token.split(".")[1];
      const base64 = base64Url.replace(/-/g, "+").replace(/_/g, "/");
      const jsonPayload = decodeURIComponent(
        atob(base64)
          .split("")
          .map(function (c) {
            return "%" + ("00" + c.charCodeAt(0).toString(16)).slice(-2);
          })
          .join("")
      );
      const payload = JSON.parse(jsonPayload);
      if (payload.email) setCustomerEmail(payload.email);
      const firstName =
        (payload.given_name || payload.name || "").split(" ")[0] || "";
      if (firstName) setCustomerName(firstName);
    } catch (e) {
      console.error("Google parse failed", e);
      setCheckoutError("Google sign-in failed");
    }
  };
