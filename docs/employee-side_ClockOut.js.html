import React, { useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";


/**
 * ClockOut component allows a cashier to end their shift.
 * 
 * It calculates the total shift duration based on the shift start time
 * stored in localStorage. The component displays the cashier's name,
 * total hours worked, and provides options to confirm clock out or cancel.
 * 
 * @component
 * @example
 * return (
 *   <ClockOut />
 * )
 */
export default function ClockOut() {
  const nav = useNavigate();

  /**
   * The name of the cashier retrieved from localStorage.
   * Defaults to "Cashier" if not found.
   * @type {string}
   */
  const cashierName = localStorage.getItem("cashierName") || "Cashier";

  /**
   * State to store the total shift duration in seconds.
   * @type {Object}
   */
  const [shiftSeconds, setShiftSeconds] = useState(0);


   /**
   * useEffect hook to calculate shift duration on component mount.
   * Retrieves the shift start time from localStorage and computes
   * the difference with the current time in seconds.
   */
  useEffect(() => {
    const start = localStorage.getItem("shiftStartTime");
    if (!start) return;
    const startTime = new Date(start).getTime();
    const now = Date.now();
    setShiftSeconds(Math.floor((now - startTime) / 1000));
  }, []);

  // Convert shiftSeconds into hours, minutes, and seconds
  const hours = Math.floor(shiftSeconds / 3600);
  const minutes = Math.floor((shiftSeconds % 3600) / 60);
  const seconds = shiftSeconds % 60;

 /**
   * Formatted shift duration in HH:MM:SS format.
   * @type {string}
   */
  const shiftDisplay =
    String(hours).padStart(2, "0") + ":" +
    String(minutes).padStart(2, "0") + ":" +
    String(seconds).padStart(2, "0");

   /**
   * Handles confirming the clock out.
   * Removes the shift start time from localStorage and navigates to login page.
   */
  const handleConfirmClockOut = () => {
    localStorage.removeItem("shiftStartTime");
    nav("/login");
  };

   /**
   * Handles canceling the clock out.
   * Navigates back to the POS page.
   */
  const handleCancel = () => {
    nav("/pos");
  };

  return (
    <div style={{
      display: "flex",
      flexDirection: "column",
      alignItems: "center",
      justifyContent: "center",
      height: "100vh",
      backgroundColor: "#f1f5f9",
      textAlign: "center"
    }}>
      <h1 style={{ fontSize: "28px", marginBottom: "8px" }}>
        Clock out as: {cashierName}?
      </h1>
      <p style={{ fontSize: "18px", marginBottom: "20px" }}>
        Hours this shift: {shiftDisplay}
      </p>
      <div style={{ display: "flex", gap: "12px" }}>
        <button
          onClick={handleConfirmClockOut}
          style={{
            backgroundColor: "#dc2626",
            color: "white",
            padding: "10px 22px",
            fontSize: "18px",
            fontWeight: "600",
            border: "none",
            borderRadius: "8px",
            cursor: "pointer",
          }}
        >
          Confirm Clock Out
        </button>
        <button
          onClick={handleCancel}
          style={{
            backgroundColor: "#e5e7eb",
            color: "#111827",
            padding: "10px 22px",
            fontSize: "18px",
            fontWeight: "500",
            border: "none",
            borderRadius: "8px",
            cursor: "pointer",
          }}
        >
          Cancel
        </button>
      </div>
    </div>
  );
}
