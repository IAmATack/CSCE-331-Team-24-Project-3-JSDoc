import React from "react";
import { Link } from 'react-router-dom';
import "./Inventory.css";
import "./Manager.css";
import Dropdown from "../components/Dropdown/Dropdown.js";
import DropdownItem from "../components/Dropdown/DropdownItem.js";
import client from './clientConnection.js';

/**
 * Inventory component: UI and logic for managing inventory items.
 *
 * - Loads inventory item names and details via `client`.
 * - Allows selecting an item, updating fields, refilling stock (single or to target),
 *   generating a restock report, and adding new inventory items.
 */
function Inventory() {
    /** Displayable field labels used by update dropdowns */
    const fields = ["Name", "Price", "Stock", "Target Stock"];

    /**
     * Inventory item list (loaded from `inventory` table)
     * @type {Array<Object>}
     */
    const [items, setItems] = React.useState([]);

    /**
     * Selected inventory item details shown in the UI
     * @type {{id:number,itemname:string,price:number,stock:number,targetStock:number}}
     */
    const [selectedItem, updateSelectedItem] = React.useState ({
        id: -1,
        itemname: "Select an Item",
        price: 0.0,
        stock: 0,
        targetStock: 0,
    });

    /** New item name (Add Item form) */
    const [newItemName, setNewItemName] = React.useState("");

    /** New item target stock (Add Item form) */
    const [newItemTargetStock, setNewItemTargetStock] = React.useState(0);

    /** New item price (Add Item form) */
    const [newItemPrice, setNewItemPrice] = React.useState(0);
    

    /**
     * Load list of inventory item names on component mount
     * @returns {void}
     */
    React.useEffect(() => {
        const fetchItems = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "inventory",
                    columns: ["itemname"],
                });
                setItems(data);
            } catch (error) {
                console.error("An error occurred:", error.message);
                setItems(["Unable to load items"]);
            }
        };

        fetchItems();
    }, []);

    /**
     * When `selectedItem.itemname` changes, fetch full details for that item
     * @returns {void}
     */
    React.useEffect(() => {
        if (!selectedItem.itemname || selectedItem.itemname === "Select an Item") return; // do nothing if no selection


        const fetchItemDetails = async () => {
            try {
                const data = await client.genericTableGet({
                    table: "inventory",
                    columns: ["id", "itemname", "price", "stock", "targetstock"],
                });

                // Filter the result for the selected Item
                const itm = data.find(e => e.itemname === selectedItem.itemname);

                if (itm) {
                    updateSelectedItem({
                        id: itm.id ?? -1,
                        itemname: itm.itemname ?? "",
                        price: itm.price ?? 0.0,
                        stock: itm.stock ?? 0,
                        targetStock: itm.targetstock ?? 0,
                    });
                }
            } catch (error) {
                console.error("Failed to fetch item details:", error.message);
            }
        };

        fetchItemDetails();
    }, [selectedItem.itemname]);

    /**
     * Select an inventory item by name; triggers detail fetch via effect.
     * @param {string} name
     * @returns {void}
     */
    const onItemSelect = (name) => {
        updateSelectedItem({
            itemname: name,
        })
    }

    /** Which field is selected for updating the current item */
    const [selectedItemField, updateSelectedItemField] = React.useState("Field to Update");

    const onItemFieldSelect = (name) => {
        updateSelectedItemField(name);
    }

    /** New value entered for the selected field */
    const [newItemValue, setNewItemValue] = React.useState("");

    /**
     * Validate and submit a single-field update for the selected inventory item.
     * @async
     * @returns {Promise<void>}
     */
    const onUpdateItemClick = async () => {
        if (!selectedItem.itemname || !selectedItemField || newItemValue === "") {
            alert("Please select an item, a field, and enter a new value");
            return;
        }


        try {
            // Map field names to database column names
            const columnMap = {
                "Name": "itemname",
                "Price": "price",
                "Stock": "stock",
                "Target Stock": "targetstock"
            };

            const column = columnMap[selectedItemField];

            if (!column) throw new Error("Invalid field selected");
            
            // Explicit type conversion if necessary
            let valueToSet;
            if (column === "price" || column === "stock" || column === "targetstock") {
                valueToSet = Number(newItemValue);
            } else {
                valueToSet = newItemValue;
            }

            await client.genericTableSet({
                table: "inventory",
                column: column,
                value: valueToSet,
                id: selectedItem.id 
            });

            alert("Item updated successfully!");

            const updatedData = await client.genericTableGet({
                table: "inventory",
                columns: ["id", "itemname", "price", "stock", "targetstock"],
                id: selectedItem.id
            });

            if (updatedData && updatedData.length > 0) {
                const itm = updatedData[0];
                updateSelectedItem({
                    id: itm.id ?? -1,
                    itemname: itm.itemname ?? "",
                    price: itm.price ?? 0.0,
                    stock: itm.stock ?? 0,
                    targetStock: itm.targetstock ?? 0,
                });
            }
        } catch (error) {
            console.error("Failed to update item:", error);
            alert("Failed to update item: " + error.message);
        }
    }

    /** Report table rows for the inventory restock report */
    const [reportTableData, setReportTableData] = React.useState([]);

    /**
     * Generate an inventory restock report (items below target stock).
     * @async
     * @returns {Promise<void>}
     */
    const onGenerateInventoryReportClick = async () => {

        try {
            const items = await client.getReport(23);
            const reportItems = items.data.filter(item => item.stock < item.targetstock);
            setReportTableData(reportItems);
        } catch (error) {
            console.error("Failed to get items:", error);
            alert("Failed to get items: " + error.message);
        }
    }

    /**
     * Refill the selected item's stock by one unit and refresh state
     * @async
     * @returns {Promise<void>}
     */
    const onRefillSingleClick = async () => {

        try {
            await client.genericTableSet({
                table: "inventory",
                column: "stock",
                value: Number(selectedItem.stock)+1,
                id: selectedItem.id 
            });

            alert("Item refilled successfully!");

            const updatedData = await client.genericTableGet({
                table: "inventory",
                columns: ["id", "itemname", "price", "stock", "targetstock"],
                id: selectedItem.id
            });

            if (updatedData && updatedData.length > 0) {
                const itm = updatedData[0];
                updateSelectedItem({
                    id: itm.id ?? -1,
                    itemname: itm.itemname ?? "",
                    price: itm.price ?? 0.0,
                    stock: itm.stock ?? 0,
                    targetStock: itm.targetstock ?? 0,
                });
            }

            if (reportTableData.includes(selectedItem)) {
                setReportTableData(oldItems =>
                    oldItems.map(item =>
                        item.itemname === updatedData.itemname ? { ...item, stock: updatedData.stock} : item
                    )
                );

            }
        } catch (error) {
            console.error("Failed to refill item:", error);
            alert("Failed to refill item: " + error.message);
        }
    }

    /**
     * Refill the provided item up to its target stock and refresh state
     * @param {Object} itemData
     * @async
     * @returns {Promise<void>}
     */
    const onRefillWholeClick = async (itemData) => {

        let valueToSet = (itemData.targetstock || itemData.targetStock)

        try {
            await client.genericTableSet({
                table: "inventory",
                column: "stock",
                value: Number(valueToSet),
                id: itemData.id 
            });

            alert("Item refilled successfully!");

            const updatedData = await client.genericTableGet({
                table: "inventory",
                columns: ["id", "itemname", "price", "stock", "targetstock"],
                id: itemData.id
            });

            console.log(selectedItem.itemname)
            if (selectedItem.itemname === itemData.itemname) {
                if (updatedData && updatedData.length > 0) {
                    const itm = updatedData[0];
                    updateSelectedItem({
                        id: itm.id ?? -1,
                        itemname: itm.itemname ?? "",
                        price: itm.price ?? 0.0,
                        stock: itm.stock ?? 0,
                        targetStock: itm.targetstock ?? 0,
                    });
                }
            }

            if (reportTableData.includes(itemData)) {
                setReportTableData(oldItems =>
                    oldItems.map(item =>
                        item.itemname === updatedData[0].itemname ? { ...item, stock: updatedData[0].stock} : item
                    )
                );

            }
        } catch (error) {
            console.error("Failed to refill item:", error);
            alert("Failed to refill item: " + error.message);
        }
    }

    /**
     * Create a new inventory item using values from the Add Item form
     * @async
     * @returns {Promise<void>}
     */
    const onAddItemClick = async () => {

        try {
            await client.genericTableCreate({
                table: "inventory",
                column: "stock",
                values: {
                    itemname: newItemName,
                    price: newItemPrice,
                    stock: 0,
                    targetstock: newItemTargetStock
                }
            });

            alert("Item refilled successfully!");
        } catch (error) {
            console.error("Failed to add item:", error);
            alert("Failed to add item: " + error.message);
        }
    }

    /** Render UI: inventory controls, report, and add-item form */
    return (
    <div>
        {/* Navigation Landmark */}
        <nav
        role="navigation"
        aria-label="Inventory navigation"
        style={{ padding: "10px", backgroundColor: "#f0f0f0" }}
        >
        <Link to="/manager" style={{ margin: "10px" }}>Manager Home</Link>
        <Link to="/employees" style={{ margin: "10px" }}>Employees</Link>
        <Link to="/inventory" style={{ margin: "10px" }}>Inventory</Link>
        <Link to="/menu" style={{ margin: "10px" }}>Menu</Link>
        <Link to="/reports" style={{ margin: "10px" }}>Reports</Link>
        </nav>

        <div className="grid-container">
        {/* ---------- ITEMS COLUMN ---------- */}
        <div className="column">
            <h2 id="inventory-items-heading">Inventory Items</h2>

            <Dropdown
            buttonText={selectedItem.itemname}
            role="menu"
            ariaLabelledBy="inventory-items-heading"
            >
            {items.map((itm, index) => (
                <DropdownItem
                key={index}
                onClick={() => onItemSelect(itm.itemname)}
                role="menuitem"
                >
                {itm.itemname}
                </DropdownItem>
            ))}
            </Dropdown>

            <progress
            value={selectedItem.stock}
            max={selectedItem.targetStock}
            aria-valuenow={selectedItem.stock}
            aria-valuemax={selectedItem.targetStock}
            aria-label={`Stock level for ${selectedItem.itemname}`}
            />

            <table role="table" aria-labelledby="inventory-items-heading">
            <tbody>
                <tr>
                <th scope="row">Name:</th>
                <td>{selectedItem.itemname}</td>
                </tr>
                <tr>
                <th scope="row">Price:</th>
                <td>{selectedItem.price}</td>
                </tr>
                <tr>
                <th scope="row">Stock:</th>
                <td>{selectedItem.stock}</td>
                </tr>
                <tr>
                <th scope="row">Target Stock:</th>
                <td>{selectedItem.targetStock}</td>
                </tr>
            </tbody>
            </table>

            <button
            className="defaultButton"
            onClick={onRefillSingleClick}
            aria-label={`Refill 1 unit for ${selectedItem.itemname} at $${selectedItem.price}`}
            >
            Refill 1 (${selectedItem.price})
            </button>

            <button
            className="defaultButton"
            onClick={() => onRefillWholeClick(selectedItem)}
            aria-label={`Refill ${selectedItem.itemname} to target stock at $${Math.max(selectedItem.targetStock - selectedItem.stock, 0) * selectedItem.price}`}
            >
            Refill to Target (${(selectedItem.targetStock - selectedItem.stock > 0) ? selectedItem.price*(selectedItem.targetStock-selectedItem.stock) : 0})
            </button>

            <Dropdown
            buttonText={selectedItemField}
            role="menu"
            ariaLabel="Select item field to update"
            >
            {fields.map((field) => (
                <DropdownItem
                key={field}
                onClick={() => onItemFieldSelect(field)}
                role="menuitem"
                >
                {field}
                </DropdownItem>
            ))}
            </Dropdown>

            <label htmlFor="new-item-value">New Value:</label>
            <input
            id="new-item-value"
            value={newItemValue}
            onChange={(e) => setNewItemValue(e.target.value)}
            />

            <button
            className="defaultButton"
            onClick={onUpdateItemClick}
            aria-label="Save updated item value"
            >
            Save
            </button>
        </div>

        {/* ---------- REPORT & ADD ITEM COLUMN ---------- */}
        <div className="column">
            <h2 id="inventory-report-heading">Inventory Restock Report</h2>

            <button
            className="defaultButton"
            onClick={onGenerateInventoryReportClick}
            aria-label="Generate Inventory Restock Report"
            >
            Generate Inventory Restock Report
            </button>

            <div className="reportTable">
            <table role="table" aria-labelledby="inventory-report-heading">
                <thead>
                <tr>
                    <th scope="col">Name</th>
                    <th scope="col">Price</th>
                    <th scope="col">Stock</th>
                    <th scope="col">Target Stock</th>
                    <th scope="col">Refill</th>
                </tr>
                </thead>
                <tbody>
                {reportTableData.map((rowItem) => (
                    <tr key={rowItem.itemname}>
                    <td>{rowItem.itemname}</td>
                    <td>{rowItem.price}</td>
                    <td>{rowItem.stock}</td>
                    <td>{rowItem.targetstock}</td>
                    <td>
                        <button
                        onClick={() => onRefillWholeClick(rowItem)}
                        aria-label={`Refill ${rowItem.itemname} to target stock at $${Math.max(rowItem.targetstock - rowItem.stock, 0) * rowItem.price}`}
                        >
                        Refill to Target (${(rowItem.targetstock - rowItem.stock > 0) ? rowItem.price*(rowItem.targetstock-rowItem.stock) : 0})
                        </button>
                    </td>
                    </tr>
                ))}
                </tbody>
            </table>
            </div>

            <h2 id="add-item-heading">Add Item</h2>
            <div className="grid-container">
            <div className="nestedColumn">
                <label htmlFor="new-item-name">Name:</label>
                <input
                id="new-item-name"
                value={newItemName}
                onChange={(e) => setNewItemName(e.target.value)}
                />

                <label htmlFor="new-item-target-stock">Target Stock:</label>
                <input
                id="new-item-target-stock"
                type="number"
                value={newItemTargetStock}
                onChange={(e) => setNewItemTargetStock(Number(e.target.value))}
                min="0"
                max="1000"
                step="1"
                />
            </div>

            <div className="nestedColumn">
                <label htmlFor="new-item-price">Price:</label>
                <input
                id="new-item-price"
                type="number"
                value={newItemPrice}
                onChange={(e) => setNewItemPrice(Number(e.target.value))}
                min="0"
                max="100"
                step=".25"
                />

                <button
                className="defaultButton"
                onClick={onAddItemClick}
                aria-label="Save new item"
                >
                Save
                </button>
            </div>
            </div>
        </div>
        </div>
    </div>
    );
}

export default Inventory;
