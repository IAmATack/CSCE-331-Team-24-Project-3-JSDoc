// Helper function to parse user agent for device info
function parseUserAgent(userAgent) {
  if (!userAgent || userAgent === 'Unknown User-Agent') {
    return 'Device: Unknown';
  }
  
  let device = 'Desktop';
  let browser = 'Unknown';
  let os = 'Unknown';
  
  // Detect device type
  if (/mobile|android|iphone|ipad|ipod|blackberry|iemobile|opera mini/i.test(userAgent)) {
    device = 'Mobile';
  } else if (/tablet|ipad|playbook|silk/i.test(userAgent)) {
    device = 'Tablet';
  }
  
  // Detect browser
  if (/chrome/i.test(userAgent) && !/edg/i.test(userAgent)) {
    browser = 'Chrome';
  } else if (/firefox/i.test(userAgent)) {
    browser = 'Firefox';
  } else if (/safari/i.test(userAgent) && !/chrome/i.test(userAgent)) {
    browser = 'Safari';
  } else if (/edg/i.test(userAgent)) {
    browser = 'Edge';
  } else if (/opera|opr/i.test(userAgent)) {
    browser = 'Opera';
  }
  
  // Detect OS
  if (/windows/i.test(userAgent)) {
    os = 'Windows';
  } else if (/macintosh|mac os x/i.test(userAgent)) {
    os = 'macOS';
  } else if (/linux/i.test(userAgent)) {
    os = 'Linux';
  } else if (/android/i.test(userAgent)) {
    os = 'Android';
  } else if (/ios|iphone|ipad|ipod/i.test(userAgent)) {
    os = 'iOS';
  }
  
  return `Device: ${device} | Browser: ${browser} | OS: ${os}`;
}

app.use(cors({
  origin: function (origin, callback) {
    // Allow requests with no origin (health checks, server-to-server requests)
    if (!origin) {
      return callback(null, true);
    }
    
    // Check if origin is in allowed list
    if (allowedOrigins.indexOf(origin) !== -1) {
      // Only log if it's a new origin or in development
      if (process.env.NODE_ENV !== 'production') {
        console.log(`✅ CORS: ${origin}`);
      }
      callback(null, true);
    } else {
      console.warn(`❌ CORS blocked: ${origin} (allowed: ${allowedOrigins.join(', ')})`);
      callback(new Error('Not allowed by CORS'));
    }
  },
  credentials: true
}));

// Log request body after JSON parsing
app.use((req, res, next) => {
  if (req.method !== 'GET' && req.body && Object.keys(req.body).length > 0) {
    // Custom compact formatter for order logs
    const formatCompact = (obj, indent = 0) => {
      const pad = '  '.repeat(indent);
      
      if (Array.isArray(obj)) {
        if (obj.length === 0) return '[]';
        // Check if it's a simple array that can fit on one line
        const simpleArrayStr = JSON.stringify(obj);
        if (simpleArrayStr.length < 80 && !obj.some(item => typeof item === 'object' && item !== null)) {
          return simpleArrayStr;
        }
        // Complex array - format each item compactly
        const items = obj.map(item => formatCompact(item, indent + 1));
        return '[\n' + items.map(item => pad + '  ' + item).join(',\n') + '\n' + pad + ']';
      }
      
      if (obj && typeof obj === 'object' && obj !== null) {
        const keys = Object.keys(obj);
        if (keys.length === 0) return '{}';
        
        // Special handling for lineItems - keep on one line with entrees
        if (keys.includes('menuID') && keys.includes('quantity')) {
          const parts = [];
          if (obj.menuID !== undefined) parts.push(`menuID: ${obj.menuID}`);
          if (obj.quantity !== undefined) parts.push(`qty: ${obj.quantity}`);
          if (obj.entree && Array.isArray(obj.entree)) {
            const entreeStr = obj.entree.map(e => 
              `{menuID: ${e.menuID}, qty: ${e.quantity || 1}}`
            ).join(', ');
            parts.push(`entree: [${entreeStr}]`);
          }
          return `{ ${parts.join(', ')} }`;
        }
        
        // Regular object - format compactly
        const pairs = keys.map(key => {
          const val = obj[key];
          if (val === null) return `${key}: null`;
          if (typeof val === 'string') return `${key}: "${val}"`;
          if (typeof val === 'number' || typeof val === 'boolean') return `${key}: ${val}`;
          if (Array.isArray(val) || (typeof val === 'object')) {
            const formatted = formatCompact(val, indent + 1);
            // If formatted value has newlines, put it on next line
            if (formatted.includes('\n')) {
              return `${key}:\n${pad}  ${formatted}`;
            }
            return `${key}: ${formatted}`;
          }
          return `${key}: ${JSON.stringify(val)}`;
        });
        
        // Small objects on one line
        const singleLine = `{ ${pairs.join(', ')} }`;
        if (singleLine.length < 100 && !singleLine.includes('\n')) {
          return singleLine;
        }
        
        // Multi-line object
        return '{\n' + pairs.map(p => pad + '  ' + p).join(',\n') + '\n' + pad + '}';
      }
      
      return JSON.stringify(obj);
    };

    const bodyStr = typeof req.body === 'object' ? formatCompact(req.body) : String(req.body);
    const maxLength = 2000;
    if (bodyStr.length > maxLength) {
      console.log(`   Body: ${bodyStr.substring(0, maxLength)}... (truncated, ${bodyStr.length} chars total)`);
    } else {
      const lines = bodyStr.split('\n');
      if (lines.length === 1) {
        console.log(`   Body: ${bodyStr}`);
      } else {
        console.log(`   Body:`);
        lines.forEach(line => console.log(`      ${line}`));
      }
    }
  }
  next();
});

/**
 * Initialize server with async startup sequence
 * 1. Initialize database schema
 * 2. Setup API endpoints
 * 3. Start listening
 * @author Danny
 */
async function startServer() {
  try {
    // Step 1: Initialize database schema
    console.log("Initializing server...");
    await initializeServerLogic();
    
    // Step 2: Setup API endpoints
    setupServerConnections(app);
    
    // Step 3: Start listening
    app.listen(port, async () => {
      console.log(`✅ Server running at ${port}`);

      try { // tests pipeline of ServerConnections -> ServerLogic -> DatabaseAccess -> DB
        const ok = await testDatabasePipeline(); 
        if (ok) console.log("✅ Database-Server pipeline OK");
        else console.error("❌ Database pipeline reported failure");
      } 
      catch (err) { console.error("❌ Server-Database crash:", err); }
    });
  } catch (err) {
    console.error("❌ Failed to start server:", err);
    process.exit(1);
  }
}

// Start the server
startServer();
