import { pool } from "./db.js";

/**
 * Special reports generator (X and Z reports).
 *
 * Responsibilities:
 * - Compute per-hour sales and order counts for current day (X report).
 * - Summarize totals since last Z reset and reset counters (Z report).
 */
class SpecialReports {
  constructor() {
    this.lastZReportTime = null;
    this.totalDayOrders = 0;
    this.hourRevenue = Object.fromEntries(
      Array.from({ length: 24 }, (_, h) => [h, 0])
    );
    this.generatedXReport = false;
  }

  /**
   * Generate X report: sales and number of orders per hour today.
   * Initializes the last Z timestamp if missing.
   * @returns {Promise<Array<{hour:number,numOrders:number,sales:number}>>}
   */
  async generateXReport() {
    console.log("Generating: X Report");
    
    if(this.lastZReportTime === null){
        const queryYesterday = await pool.query("SELECT (NOW() - INTERVAL '1 DAY')::timestamp AS yesterdaydate;");
        const yesterday = queryYesterday.rows[0].yesterdaydate;
        console.log("   Yesterday timestamp from DB:", yesterday, typeof yesterday);
        
        // Handle both Date objects and timestamp strings from PostgreSQL
        if (yesterday instanceof Date) {
          this.lastZReportTime = yesterday;
        } else if (typeof yesterday === 'string') {
          this.lastZReportTime = new Date(yesterday);
        } else {
          // Fallback to 24 hours ago
          this.lastZReportTime = new Date(Date.now() - 24 * 60 * 60 * 1000);
        }
        
        console.log("   Last Z Report Time set to:", this.lastZReportTime);
    }
    
    // Convert Date object to PostgreSQL timestamp string format
    let lastZReportTimeStr;
    if (this.lastZReportTime instanceof Date) {
      lastZReportTimeStr = this.lastZReportTime.toISOString().slice(0, 19).replace('T', ' ');
    } else if (typeof this.lastZReportTime === 'string') {
      lastZReportTimeStr = this.lastZReportTime;
    } else {
      // Fallback
      lastZReportTimeStr = new Date(Date.now() - 24 * 60 * 60 * 1000).toISOString().slice(0, 19).replace('T', ' ');
    }
    
    console.log("   Using timestamp for query:", lastZReportTimeStr);
    
    const query = await pool.query(`
        SELECT 
            DATE_PART('hour', timeOfOrder) AS orderhour,
            COUNT(*) AS numorders,
            SUM(price) AS totalrevenue
        FROM orders o
        WHERE o.timeOfOrder::date=NOW()::date
        AND o.timeOfOrder > $1::timestamp
        GROUP BY orderhour
        ORDER BY orderhour;
    `, [lastZReportTimeStr]);
    

    const numOrdersMap = new Map();
    const salesMap = new Map();
    for(let i = 0; i < query.rows.length; i++){
      numOrdersMap.set(query.rows[i].orderhour, query.rows[i].numorders);
      salesMap.set(query.rows[i].orderhour, query.rows[i].totalrevenue);
    }

    const xReport = query.rows.map(row => {
        const hour = parseInt(row.orderhour, 10);
        const numOrders = parseInt(row.numorders, 10);
        const sales = parseFloat(row.totalrevenue);

        // Update running totals
        this.totalDayOrders += numOrders;
        this.hourRevenue[hour] = sales;

        return { hour, numOrders, sales };
    });

    this.generatedXReport = true;
    console.log("Generated: X Report");

    return xReport;
  }

  /**
   * Generate Z report: totals since last X report and reset state.
   * @returns {Promise<Array<{totalOrders:number,totalRevenue:number}>>}
   */
  async generateZReport() {
    console.log("Generating: Z Report");

    if(this.generatedXReport === false){
      console.log("Error: Generate a X Report first");
      return null;
    }

    // Calculate total revenue
    let totalRevenue = 0;
    for (const rev of Object.values(this.hourRevenue)) {
      totalRevenue += rev;
    }

    // Round to 2 decimal places
    const roundedRevenue = Math.round(totalRevenue * 100) / 100;

    // Create Z Report data
   const zReport = [
        {
            totalOrders: this.totalDayOrders,
            totalRevenue: roundedRevenue
        }
    ];

    console.log("Z Report:", zReport);

    // Reset running totals
    this.totalDayOrders = 0;
    for (const hour in this.hourRevenue) {
      this.hourRevenue[hour] = 0;
    }
    this.generatedXReport = false;

    // Save the Z report timestamp
    this.lastZReportTime = new Date();

    return zReport;
  }
}

export default SpecialReports;
