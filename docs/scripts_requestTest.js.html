// Allow overriding API base via CLI arg --base <url> or env API_BASE_URL
function getBaseFromArgs() {
  const idx = process.argv.indexOf('--base');
  if (idx !== -1 && process.argv[idx + 1]) {
    return process.argv[idx + 1];
  }
  return process.env.API_BASE_URL || null;
}

const client = new ClientConnection(getBaseFromArgs() || undefined);
// Parse an optional --order <id> CLI arg
function getOrderIdFromArgs() {
  const idx = process.argv.indexOf('--order');
  if (idx !== -1 && process.argv[idx + 1]) {
    const v = Number(process.argv[idx + 1]);
    return Number.isFinite(v) ? v : null;
  }
  return null;
}

async function runCreateOrderTests() {
  const testEmail = 'lindanny75@tamu.edu';
  
  console.log('📦 Creating order with email:', testEmail);
  const order = await client.createOrder({
    name: 'Test Customer',
    paymentType: 'Card',
    cashierID: 1,
    email: testEmail,
    wantsReceiptEmail: false,
    lineItems: [
      {
        menuID: 1,
        quantity: 1,
        entree: [
          { menuID: 9, quantity: 1 },
          { menuID: 20, quantity: 1 }
        ]
      },
      {
        menuID: 27,
        quantity: 1
      }
    ]
  }, testEmail, false);
  
  console.log('\n✅ Order created:');
  console.log(JSON.stringify(order, null, 2));
  
  // Check if order is associated with customer
  console.log('\n🔍 Checking order association...');
  const orderDetails = await client.genericTableGet({ 
    table: 'orders', 
    id: order.orderID 
  });
  
  if (orderDetails && orderDetails.length > 0) {
    const orderData = orderDetails[0];
    console.log('📋 Order details:');
    console.log(`   Order ID: ${orderData.id}`);
    console.log(`   Customer ID: ${orderData.customerid || 'NULL (not associated)'}`);
    console.log(`   Price: $${orderData.price}`);
    
    if (orderData.customerid) {
      console.log('   ✅ Order is associated with customer!');
      
      // Check customer details
      console.log('\n👤 Checking customer details...');
      const customerDetails = await client.genericTableGet({ 
        table: 'customers', 
        id: orderData.customerid 
      });
      
      if (customerDetails && customerDetails.length > 0) {
        const customer = customerDetails[0];
        console.log('📧 Customer information:');
        console.log(`   Customer ID: ${customer.id}`);
        console.log(`   Email: ${customer.email}`);
        console.log(`   Points: ${customer.points || 0}`);
        console.log(`   Created At: ${customer.createdat}`);
        
        if (customer.email === testEmail) {
          console.log('   ✅ Customer email matches!');
        }
        
        const expectedPoints = Math.ceil(Number(orderData.price));
        if (customer.points >= expectedPoints) {
          console.log(`   ✅ Customer has points (${customer.points} points, expected at least ${expectedPoints})`);
        } else {
          console.log(`   ⚠️ Customer points may be incorrect (${customer.points} points, expected at least ${expectedPoints})`);
        }
      } else {
        console.log('   ❌ Customer not found!');
      }
    } else {
      console.log('   ❌ Order is NOT associated with a customer!');
    }
  } else {
    console.log('   ❌ Order not found!');
  }
  
  // Check if customer exists in customer list
  console.log('\n📋 Checking customer list...');
  const allCustomers = await client.genericTableGet({ table: 'customers' });
  const customerInList = allCustomers.find(c => c.email === testEmail);
  
  if (customerInList) {
    console.log('✅ Customer found in customer list:');
    console.log(JSON.stringify(customerInList, null, 2));
  } else {
    console.log('❌ Customer NOT found in customer list!');
    console.log('Available customers:', allCustomers.map(c => c.email).join(', '));
  }
}

/**
 * Authenticate as a manager to get database access
 * Uses the same credentials as runLoginTests for consistency
 */
async function authenticate(client) {
  console.log('🔐 Authenticating as manager...');
  const candidates = [
    { username: 'manager', password: 'pass', role: 1 }
  ];

  for (const creds of candidates) {
    try {
      const loginResult = await client.validateEmployeeLogin(creds);
      if (loginResult && loginResult.success) {
        console.log('✅ Authentication successful:', loginResult.employee?.name || loginResult.employee?.username);
        return loginResult;
      } else {
        console.log(`⚠️ Auth failed for ${creds.username}: ${loginResult?.message || 'Unknown error'}`);
      }
    } catch (e) {
      console.log(`⚠️ Auth request error for ${creds.username}: ${e.message || e}`);
    }
  }

  const err = new Error('Authentication failed for all candidates');
  err.name = 'AuthError';
  throw err;
}
