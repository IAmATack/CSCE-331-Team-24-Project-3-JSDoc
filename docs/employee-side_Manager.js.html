import { Link, useNavigate } from 'react-router-dom';
import "./Manager.css";
import LogoutButton from '../components/LogoutButton/logoutButton.js';

/**
 * Manager view component.
 *
 * Renders manager navigation links and quick action buttons for common views
 * (Customer Kiosk, Kitchen View, Menu Board). Uses `useNavigate` for
 * programmatic navigation.
 *
 * @returns {JSX.Element}
 */
function Manager() {
    /** Programmatic navigation helper from react-router */
    const navigate = useNavigate();

    /**
     * Navigate to the public/customer kiosk view
     * @returns {void}
     */
    const navToKiosk = () => {
        navigate('/kiosk', { replace: true });
    }

    /**
     * Navigate to the kitchen view
     * @returns {void}
     */
    const navToKitchenView = () => {
        navigate('/kitchenview', { replace: true });
    }

    /**
     * Navigate to the menu board view
     * @returns {void}
     */
    const navToMenuBoard = () => {
        navigate('/menuboard', { replace: true });
    }

    return (
  <div>
    {/* Navigation */}
    <nav
      style={{ padding: '12px 16px', backgroundColor: '#f7f7f7', borderBottom: '1px solid #e5e7eb' }}
      role="navigation"
      aria-label="Main navigation"
    >
      <Link to="/manager" style={{ margin: '10px' }}>Manager Home</Link>
      <Link to="/employees" style={{ margin: '10px' }}>Employees</Link>
      <Link to="/inventory" style={{ margin: '10px' }}>Inventory</Link>
      <Link to="/menu" style={{ margin: '10px' }}>Menu</Link>
      <Link to="/reports" style={{ margin: '10px' }}>Reports</Link>
    </nav>

    {/* Page Heading */}
    <h2 id="manager-welcome-heading" style={{ margin: '16px 0' }}>
      Welcome!
    </h2>

    {/* Action Buttons */}
    <div
      className="buttonRow"
      role="region"
      aria-labelledby="manager-welcome-heading"
    >
      <LogoutButton aria-label="Logout from manager dashboard">Logout</LogoutButton>

      <button
        className="defaultButton"
        onClick={navToKiosk}
        aria-label="Go to customer kiosk"
      >
        🍚 Customer Kiosk
      </button>

      <button
        className="defaultButton"
        onClick={navToKitchenView}
        aria-label="Go to kitchen view"
      >
        👩‍🍳 Kitchen View
      </button>

      <button
        className="defaultButton"
        onClick={navToMenuBoard}
        aria-label="Go to menu board"
      >
        🧾 Menu Board
      </button>
    </div>
  </div>
);

}

export default Manager;
